This implementation runs in about 0.6 s on a random graph with 10^6 nodes and 5×10^6 edges (measured below). It uses a hand-written binary min-heap of `(distance, node)` tuples with **lazy deletion**: a node is not moved within the heap when its distance improves. Instead a new entry is pushed, and outdated entries are skipped when they are popped (`d > dist[u]`). This avoids maintaining a node-to-heap-position map. The heap can hold up to one entry per edge, which costs some memory, but in practice this is usually the fastest variant of Dijkstra's algorithm with a binary heap.

```julia
# Binary min-heap of (key, node) pairs ordered by key.
function heappush!(h::Vector{Tuple{W,Int}}, x::Tuple{W,Int}) where {W}
    push!(h, x)
    i = length(h)
    while i > 1
        p = i >> 1
        first(h[p]) <= first(x) && break
        h[i] = h[p]
        i = p
    end
    h[i] = x
    return h
end

function heappop!(h::Vector{Tuple{W,Int}}) where {W}
    top = h[1]
    x = pop!(h)
    n = length(h)
    if n > 0
        i = 1
        while true
            c = 2i
            c > n && break
            if c < n && first(h[c+1]) < first(h[c])
                c += 1
            end
            first(x) <= first(h[c]) && break
            h[i] = h[c]
            i = c
        end
        h[i] = x
    end
    return top
end

weighttype(::AbstractVector{<:AbstractVector{<:Tuple{Integer,W}}}) where {W<:Real} = W

infinity(::Type{W}) where {W<:Real} = typemax(W)
infinity(::Type{W}) where {W<:AbstractFloat} = W(Inf)

"""
    dijkstra(adj, src) -> (dist, prev)

Single-source shortest paths. `adj[u]` holds `(v, w)` pairs for edges `u → v`
with weight `w ≥ 0`. Unreachable nodes get `dist = typemax(W)` (`Inf` for
floats) and `prev = 0`; `prev[src] == 0` as well.
"""
function dijkstra(adj::AbstractVector{<:AbstractVector{<:Tuple{Integer,Real}}}, src::Integer)
    Base.require_one_based_indexing(adj)
    n = length(adj)
    1 <= src <= n || throw(ArgumentError("source $src is not a node (1:$n)"))
    W = weighttype(adj)
    inf = infinity(W)
    dist = fill(inf, n)
    prev = zeros(Int, n)
    dist[src] = zero(W)
    heap = Tuple{W,Int}[(zero(W), Int(src))]
    while !isempty(heap)
        d, u = heappop!(heap)
        d > dist[u] && continue          # stale entry: u was already settled with a smaller distance
        for (v, w) in adj[u]
            w < 0 && throw(ArgumentError("negative weight $w on edge $u → $v"))
            nd = d + w
            if nd < dist[v]
                dist[v] = nd
                prev[v] = u
                heappush!(heap, (nd, Int(v)))
            end
        end
    end
    return dist, prev
end

"""
    path(prev, target) -> Vector{Int}

Nodes from the source to `target` following `prev`. Returns `[target]` when
`target` is the source *or* unreachable; check `dist[target]` to tell them apart.
"""
function path(prev::AbstractVector{<:Integer}, target::Integer)
    p = Int[]
    v = Int(target)
    while v != 0
        push!(p, v)
        length(p) > length(prev) && throw(ArgumentError("prev contains a cycle"))
        v = prev[v]
    end
    return reverse!(p)
end
```

Notes:
- **Weight type.** `W` is taken from the adjacency element type, so `Int`, `Float32`, `Float64` and `Rational` weights each get distances of that same type. Unreachable nodes get `Inf` for floating-point weights and `typemax(W)` otherwise (for example `typemax(Int)`, or `1//0` for `Rational`).
- **Unsupported weight types.** If the adjacency has an abstract weight type (for example `Vector{Vector{Tuple{Int,Real}}}`), or a weight type without `typemax` such as `BigInt`, the call fails with a `MethodError` rather than producing a guessed result.
- **Integer overflow.** With integer weights, a sum of large weights can overflow, as with any `Int` arithmetic.
- **Negative weights** are detected when the edge is scanned, and an `ArgumentError` is thrown.
- **`path(prev, t)`** returns `[t]` both for the source and for an unreachable `t`, because `prev` alone cannot tell the two apart. Check `isfinite(dist[t])` (or `dist[t] < typemax`) first.

using Test, Random
function bellman(adj, src)
    n = length(adj); W = weighttype(adj)
    d = fill(infinity(W), n); d[src] = 0
    for _ in 1:n, u in 1:n, (v, w) in adj[u]
        d[u] < infinity(W) && d[u] + w < d[v] && (d[v] = d[u] + w)
    end
    d
end
@testset "dijkstra" begin
    adj = [[(2, 7), (3, 9), (6, 14)], [(1, 7), (3, 10), (4, 15)], [(1, 9), (2, 10), (4, 11), (6, 2)],
           [(2, 15), (3, 11), (5, 6)], [(4, 6), (6, 9)], [(1, 14), (3, 2), (5, 9)], Tuple{Int,Int}[]]
    dist, prev = dijkstra(adj, 1)
    @test dist == [0, 7, 9, 20, 20, 11, typemax(Int)]
    @test path(prev, 5) == [1, 3, 6, 5]
    @test path(prev, 1) == [1]
    @test prev[7] == 0
    fadj = [[(2, 0.5)], [(3, 0.25)], Tuple{Int,Float64}[]]
    d, p = dijkstra(fadj, 1); @test d == [0.0, 0.5, 0.75]
    d, p = dijkstra(fadj, 3); @test d[1] == Inf && p == [0, 0, 0]
    radj = [[(2, 1//3)], [(3, 1//6)], Tuple{Int,Rational{Int}}[]]
    d, _ = dijkstra(radj, 1); @test d == [0, 1//3, 1//2] && eltype(d) == Rational{Int}
    f32 = [[(2, 1.5f0)], Tuple{Int,Float32}[]]
    d, _ = dijkstra(f32, 1); @test eltype(d) == Float32
    @test_throws ArgumentError dijkstra([[(2, -1)], Tuple{Int,Int}[]], 1)
    @test_throws ArgumentError dijkstra(adj, 8)
    rng = Xoshiro(1)
    for trial in 1:50
        n = rand(rng, 1:30)
        g = [Tuple{Int,Float64}[(rand(rng, 1:n), rand(rng)) for _ in 1:rand(rng, 0:4)] for _ in 1:n]
        s = rand(rng, 1:n)
        d, p = dijkstra(g, s)
