`parse_csv_line` scans the line's bytes (`codeunits`) once and passes over UTF-8 sequences without decoding them. This is correct because the bytes it looks for (`"`, the delimiter, `\r`) are all ASCII, and in UTF-8 an ASCII byte never occurs inside a multi-byte character. Field bytes are copied unchanged, so every field returned is valid UTF-8 whenever the input is.

```julia
"""
    parse_csv_line(line::AbstractString; delim=',') -> Vector{String}

Split one CSV record per RFC 4180. A field that starts with `"` is quoted: it
may contain `delim`, newlines and `""` (an escaped `"`), and must be followed by
`delim` or the end of the line. A `"` inside an unquoted field is kept
literally. A single trailing `\\r` (from CRLF line endings) is ignored.
Throws `ArgumentError` for an unterminated quoted field or for characters
between a closing quote and the next delimiter.
"""
function parse_csv_line(line::AbstractString; delim::AbstractChar=',')
    (isascii(delim) && delim != '"' && delim != '\r') ||
        throw(ArgumentError("delimiter must be an ASCII character other than '\"' and '\\r', got $(repr(delim))"))
    s = String(line)
    cu = codeunits(s)
    n = length(cu)
    n > 0 && cu[n] == UInt8('\r') && (n -= 1)
    d = UInt8(delim)
    q = UInt8('"')
    out = String[]
    buf = UInt8[]
    i = 1                                  # start of the current field
    while true
        if i <= n && cu[i] == q            # quoted field
            qstart = i
            i += 1
            while true
                i > n && throw(ArgumentError("unterminated quoted field starting at byte $qstart"))
                c = cu[i]
                if c == q
                    if i < n && cu[i+1] == q
                        push!(buf, q)      # "" -> "
                        i += 2
                    else
                        i += 1             # closing quote
                        break
                    end
                else
                    push!(buf, c)
                    i += 1
                end
            end
            push!(out, String(buf))        # takes buf's bytes and leaves buf empty
            i > n && return out
            cu[i] == d || throw(ArgumentError("unexpected character after closing quote at byte $i"))
            i += 1
        else                               # unquoted field: copy bytes up to the delimiter
            j = i
            while j <= n && cu[j] != d
                j += 1
            end
            push!(out, j == i ? "" : GC.@preserve(s, unsafe_string(pointer(s, i), j - i)))
            j > n && return out
            i = j + 1
        end
    end
end
```

Design notes:
- **Unquoted fields** are copied with a single allocation, using `unsafe_string(pointer(s, i), len)`. `GC.@preserve` keeps `s` alive while its pointer is in use.
- **Quoted fields** collect their bytes in `buf`, which handles `""` → `"`. `String(buf)` takes over `buf`'s memory without copying and leaves `buf` empty, ready for the next quoted field.
- **Strictness.** An unterminated quote, or anything other than the delimiter after a closing quote (`"ab"x`), throws an `ArgumentError` that gives the byte position. A stray `"` in the middle of an unquoted field (`a"b`) is kept as a literal character, which is lenient but common in practice.
- **Line endings.** A trailing `\r` is dropped, because `eachline` leaves it on CRLF files. A quoted field may contain a newline when the caller passes the whole record; `eachline` itself splits at newlines, so you would need to rejoin lines while a quote is open.
- **Non-`String` input** (for example a `SubString`) is converted to `String` once at the start.

### Tests

```julia
using Test
@testset "parse_csv_line" begin
    @test parse_csv_line("a,b,c") == ["a", "b", "c"]
    @test parse_csv_line("") == [""]
    @test parse_csv_line(",") == ["", ""]
    @test parse_csv_line("a,") == ["a", ""]
    @test parse_csv_line("\"a,b\",c") == ["a,b", "c"]
    @test parse_csv_line("\"he said \"\"hi\"\"\",x") == ["he said \"hi\"", "x"]
    @test parse_csv_line("\"\"") == [""]
    @test parse_csv_line("\"\",\"\"") == ["", ""]
    @test parse_csv_line("\"a\",") == ["a", ""]
    @test parse_csv_line("\"line1\nline2\",z") == ["line1\nline2", "z"]
    @test parse_csv_line("a\"b,c") == ["a\"b", "c"]
    @test parse_csv_line("a,b\r") == ["a", "b"]
    @test parse_csv_line("a;b,c;d"; delim=';') == ["a", "b,c", "d"]
    @test parse_csv_line("a\tb"; delim='\t') == ["a", "b"]
    @test parse_csv_line("héllo,\"wörld, ok\",日本") == ["héllo", "wörld, ok", "日本"]
    r = parse_csv_line("ä,\"ö\"\"ü\"")
    @test r == ["ä", "ö\"ü"] && all(isvalid, r)
    @test parse_csv_line(SubString("xx,a,b", 4)) == ["a", "b"]
    @test_throws "unterminated quoted field starting at byte 3" parse_csv_line("a,\"bc")
    @test_throws ArgumentError parse_csv_line("\"abc\"\"")
    @test_throws "unexpected character after closing quote at byte 5" parse_csv_line("\"ab\"x,y")
    @test_throws ArgumentError parse_csv_line("a"; delim='"')
    @test_throws ArgumentError parse_csv_line("a"; delim='é')
    # round trip against a simple writer
    esc(f) = any(c -> c in (',', '"', '\n'), f) || isempty(f) && rand(Bool) ? "\"" * replace(f, "\"" => "\"\"") * "\"" : f
    for _ in 1:2000
        fields = [String(rand(['a', 'é', ',', '"', ' ', '\n', 'z'], rand(0:5))) for _ in 1:rand(1:6)]
        @test parse_csv_line(join(esc.(fields), ",")) == fields
    end
end
```

All 2,022 checks pass. These include 2,000 random round trips through a minimal CSV writer, with fields containing `,`, `"`, newlines, `é` and empty strings.

### Benchmark (measured, BenchmarkTools, Julia 1.13)

10^6 lines of 6 fields each, where about 20% of the fields are quoted with an embedded delimiter and `""`:

```julia
lines = [join((rand(rng) < 0.2 ? "\"q, $(rand(rng, 1:999)) \"\"x\"\"\"" : string(rand(rng, 1:10^6))
               for _ in 1:6), ",") for _ in 1:10^6]
# e.g. "\"q, 349 \"\"x\"\"\",628265,192809,780520,167713,452809"
function parse_all(lines)
    t = 0
    for l in lines
        t += length(parse_csv_line(l))
    end
    t
end
@btime parse_all($lines)     # 262.2 ms (13480943 allocations: 539.00 MiB)
```

That is 262 ns per line, comfortably under one second for 10^6 lines. The remaining cost is mostly allocation: the result vector plus one `String` per field. If you parse in a loop and want to reuse the output vector, a variant `parse_csv_line!(out, line)` that starts with `empty!(out)` would remove one allocation per line.
