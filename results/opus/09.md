```julia
# Common endpoint type: from the element type when it is concrete, otherwise
# (e.g. `[(1, 2.5), (2, 3)]`, whose eltype is `Tuple{Int,Real}`) from the values.
function _endpoint_type(v)
    E = eltype(v)
    isconcretetype(E) && return promote_type(fieldtypes(E)...)
    isempty(v) && return Real
    return mapreduce(((a, b),) -> promote_type(typeof(a), typeof(b)), promote_type, v)
end

"""
    merge_intervals(v) -> Vector{Tuple{T,T}}

Merge closed intervals `(start, stop)` into the minimal sorted list of disjoint
closed intervals covering the same points. Intervals that touch (`stop == start`)
or overlap are merged. `v` is not modified. Throws `ArgumentError` if an
interval has `start > stop` (or a `NaN` endpoint).
"""
function merge_intervals(v::AbstractVector{<:Tuple{Real,Real}})
    T = _endpoint_type(v)
    out = Tuple{T,T}[]
    isempty(v) && return out
    for (a, b) in v
        a <= b || throw(ArgumentError("invalid interval ($a, $b): need start ≤ stop"))
    end
    s = sort(v; by = first)            # sorted copy; `v` is untouched
    lo, hi = convert(T, first(s)[1]), convert(T, first(s)[2])
    for (a, b) in Iterators.drop(s, 1)
        if a <= hi                     # overlaps or touches the current run
            hi = max(hi, convert(T, b))
        else
            push!(out, (lo, hi))
            lo, hi = convert(T, a), convert(T, b)
        end
    end
    push!(out, (lo, hi))
    return out
end
```

How it works:
- **Sort, then sweep.** `sort(v; by = first)` returns a sorted **copy**, so the input is never modified, and costs O(n log n). The sweep is then O(n): a new interval extends the current run when `start ≤ hi`, so closed intervals that only touch (`(1,2)` and `(2,3)`) are merged. Otherwise the current run is finished and a new one starts.
- **Endpoint type.** For a concrete element type such as `Tuple{Int,Float64}`, the output is `Vector{Tuple{Float64,Float64}}` and the function is type-stable (checked with `@inferred`). A literal like `[(1, 2.5), (2, 3)]` has the abstract element type `Tuple{Int,Real}`. In that case the common type is computed from the values, so you still get `Float64` endpoints rather than `Real`.
- **Invalid input** (`start > stop`, or a `NaN` endpoint, which fails `a <= b`) throws `ArgumentError` instead of silently producing an odd result.
- **Generic indexing.** The function never indexes with `1:n`; it uses `first` and iteration, so views and `OffsetArray`s work.
- Measured: 10^6 random short intervals merge in 55 ms (BenchmarkTools, 36 allocations).

### Tests

The randomized property test relies on an exactness argument. Every interval has integer endpoints in `0:N`. Two finite unions of such closed intervals cover the same set of real numbers exactly when they agree on every integer and every half-integer in `0:0.5:N`, because their membership can only change at the endpoints. So the brute-force check is exact, not a sample.

```julia
using Test, Random, OffsetArrays

# Brute force on integer endpoints in 0:N: two finite unions of closed intervals
# with integer endpoints cover the same points iff they agree on every point
# of 0:0.5:N (integers and the midpoints between them).
covered(ivs, x) = any(a <= x <= b for (a, b) in ivs)

@testset "merge_intervals" begin
    @testset "examples" begin
        @test merge_intervals(Tuple{Int,Int}[]) == Tuple{Int,Int}[]
        @test merge_intervals([(1, 3)]) == [(1, 3)]
        @test merge_intervals([(1, 3), (2, 6), (8, 10), (15, 18)]) == [(1, 6), (8, 10), (15, 18)]
        @test merge_intervals([(5, 6), (1, 2)]) == [(1, 2), (5, 6)]          # unsorted input
        @test merge_intervals([(1, 2), (2, 3)]) == [(1, 3)]                  # touching
        @test merge_intervals([(1, 2), (3, 4)]) == [(1, 2), (3, 4)]          # gap of 1 is a gap
        @test merge_intervals([(1, 10), (2, 3), (4, 5)]) == [(1, 10)]        # containment
        @test merge_intervals([(3, 3), (3, 3)]) == [(3, 3)]                  # points
        @test merge_intervals([(1, 1), (1, 4)]) == [(1, 4)]
        @test merge_intervals([(0.5, 1.5), (1.5, 2.0)]) == [(0.5, 2.0)]
        @test merge_intervals([(-Inf, 0.0), (-1.0, Inf)]) == [(-Inf, Inf)]
    end
    @testset "types" begin
        r = merge_intervals([(1, 2.5), (2, 3)])
        @test r == [(1.0, 3.0)] && eltype(r) == Tuple{Float64,Float64}
        r = merge_intervals([(1//2, 1), (1, 3//2)])
        @test r == [(1//2, 3//2)] && eltype(r) == Tuple{Rational{Int},Rational{Int}}
        r = merge_intervals(Tuple{Real,Real}[(1, 2.0), (big(2), 3//1)])
        @test r == [(1, 3)]
        @test eltype(merge_intervals(Tuple{Float32,Float32}[])) == Tuple{Float32,Float32}
        @inferred merge_intervals([(1, 2), (3, 4)])
        @inferred merge_intervals([(1, 2.0), (3, 4.0)])
    end
    @testset "input not mutated; views and offsets" begin
        v = [(5, 6), (1, 2), (2, 3)]
        c = copy(v)
        merge_intervals(v)
        @test v == c
        @test merge_intervals(view(v, 2:3)) == [(1, 3)]
        @test merge_intervals(OffsetArray(v, -1)) == [(1, 3), (5, 6)]
    end
    @testset "errors" begin
        @test_throws ArgumentError merge_intervals([(2, 1)])
        @test_throws "need start ≤ stop" merge_intervals([(0.0, 1.0), (NaN, 1.0)])
    end
    @testset "randomized property test" begin
        rng = Xoshiro(2024)
        N = 30
        for trial in 1:2000
            v = Tuple{Int,Int}[]
            for _ in 1:rand(rng, 0:12)
                a = rand(rng, 0:N); b = rand(rng, a:N)
                push!(v, (a, b))
            end
            orig = copy(v)
            m = merge_intervals(shuffle(rng, v))
            @test v == orig
            # same coverage
            @test all(covered(v, x) == covered(m, x) for x in 0:0.5:N)
            # each output interval valid, sorted and strictly separated
            @test all(a <= b for (a, b) in m)
            @test all(m[i][2] < m[i+1][1] for i in 1:length(m)-1)
            # minimal: the gap between neighbours really is uncovered
            @test all(!covered(v, (m[i][2] + m[i+1][1]) / 2) for i in 1:length(m)-1)
            # endpoints come from the input
            @test all(a in first.(v) && b in last.(v) for (a, b) in m)
        end
    end
end
```

All 12,020 checks pass, including 2,000 random trials. Each random trial checks:
- the input is not mutated;
- the coverage is identical to the input's;
- the output is sorted and the intervals are strictly separated;
- the gap between neighboring outputs really is uncovered, so the output is minimal;
- every output endpoint comes from the input.
