```julia
"""
    Interval{T<:Real}(lo, hi)

Closed interval `[lo, hi]`. Arithmetic uses ordinary (round-to-nearest)
floating-point operations, without outward rounding.
"""
struct Interval{T<:Real} <: Real
    lo::T
    hi::T
    function Interval{T}(lo, hi) where {T<:Real}
        l, h = convert(T, lo), convert(T, hi)
        l <= h || throw(ArgumentError("Interval requires lo ≤ hi, got lo = $l, hi = $h"))
        return new{T}(l, h)
    end
end

Interval(lo::T, hi::T) where {T<:Real} = Interval{T}(lo, hi)
Interval(lo::Real, hi::Real) = Interval(promote(lo, hi)...)
Interval(x::Real) = Interval(x, x)
Interval{T}(x::Real) where {T<:Real} = Interval{T}(x, x)
Interval{T}(x::Interval) where {T<:Real} = Interval{T}(x.lo, x.hi)

Base.convert(::Type{Interval{T}}, x::Real) where {T<:Real} = Interval{T}(x)

Base.promote_rule(::Type{Interval{T}}, ::Type{S}) where {T<:Real,S<:Real} = Interval{promote_type(T, S)}
Base.promote_rule(::Type{Interval{T}}, ::Type{Interval{S}}) where {T<:Real,S<:Real} = Interval{promote_type(T, S)}
# Base claims promote_rule(BigFloat, <:Real) = BigFloat; this keeps the interval.
Base.promote_rule(::Type{BigFloat}, ::Type{Interval{T}}) where {T<:Real} = Interval{promote_type(BigFloat, T)}

Base.:+(a::Interval, b::Interval) = Interval(a.lo + b.lo, a.hi + b.hi)
Base.:-(a::Interval, b::Interval) = Interval(a.lo - b.hi, a.hi - b.lo)
Base.:-(a::Interval) = Interval(-a.hi, -a.lo)

function Base.:*(a::Interval, b::Interval)
    p1, p2, p3, p4 = a.lo * b.lo, a.lo * b.hi, a.hi * b.lo, a.hi * b.hi
    return Interval(min(p1, p2, p3, p4), max(p1, p2, p3, p4))
end

function Base.:/(a::Interval, b::Interval)
    b.lo <= 0 <= b.hi && throw(DivideError())
    q1, q2, q3, q4 = a.lo / b.lo, a.lo / b.hi, a.hi / b.lo, a.hi / b.hi
    return Interval(min(q1, q2, q3, q4), max(q1, q2, q3, q4))
end

Base.in(x::Real, a::Interval) = a.lo <= x <= a.hi
Base.in(b::Interval, a::Interval) = a.lo <= b.lo && b.hi <= a.hi   # containment

Base.show(io::IO, a::Interval) = print(io, "[", a.lo, ", ", a.hi, "]")
```

### How it fits together

- **Constructors.** The inner constructor `Interval{T}(lo, hi)` converts both endpoints to `T` and rejects `lo > hi`. It also rejects `NaN`, because `NaN <= x` is `false`. The outer constructors only work out `T`, so `Interval(1, 2.0)` becomes an `Interval{Float64}`, and they then pass control to the inner constructor.
- **Promotion.** `Interval` is a subtype of `Real`, so Base's generic fallbacks (`+(x::Number, y::Number) = +(promote(x, y)...)`, and similarly for the other operators) already route mixed operations through `promote`. Only the rules need to be defined:
  - `promote_rule(Interval{T}, S) = Interval{promote_type(T, S)}`. This is why `Interval(1, 2) + 1.5 === Interval(2.5, 3.5)` (an `Interval{Float64}`), and `Interval(1, 2) * Interval(0.5, 1.0) === Interval(0.5, 2.0)`.
  - The `BigFloat` rule is needed because Base declares `promote_rule(::Type{BigFloat}, ::Type{<:Real}) = BigFloat`. Without the override the two rules disagree, and in testing both `big(1.0) + Interval(1, 2)` and `Interval(1, 2) + big(1.0)` ended in a `StackOverflowError`.
- **Arithmetic** uses the standard endpoint formulas. Multiplication and division take the minimum and maximum of the four endpoint products or quotients. `a / b` throws `DivideError` whenever `0 ∈ b`, including when zero is an endpoint, as in `Interval(0, 1)`. Integer intervals divide into `Float64` intervals, consistent with `1/2 isa Float64`. `Rational` intervals stay exact.
- **`in`.** `x in a` tests membership. For two intervals, `b in a` means `b` is contained in `a`.
- **No allocation.** `Interval{Float64}` is an immutable `isbits` struct holding two `Float64`s, and all operations are type-stable, so they are done in registers. The test checks this with `@allocated == 0` for each operator and for a compound expression.
- **Rounding.** Endpoints use round-to-nearest, not outward rounding. That is enough for ordinary interval bookkeeping, but it does not give a rigorous enclosure; for that, use IntervalArithmetic.jl.

### Tests

```julia
using Test
@testset "Interval" begin
    a = Interval(1, 2)
    @test a isa Interval{Int}
    @test Interval(1, 2.0) isa Interval{Float64}
    @test_throws ArgumentError Interval(2, 1)
    @test_throws "lo ≤ hi" Interval(2.0, 1.0)
    @test_throws ArgumentError Interval(NaN, 1.0)

    r = a + 1.5
    @test r === Interval(2.5, 3.5)
    @test 1.5 + a === Interval(2.5, 3.5)
    r = a * Interval(0.5, 1.0)
    @test r === Interval(0.5, 2.0)
    @test a + a === Interval(2, 4)
    @test a - Interval(0, 1) === Interval(0, 2)
    @test -a === Interval(-2, -1)
    @test Interval(-1, 2) * Interval(-3, 1) === Interval(-6, 3)
    @test Interval(1, 2) / Interval(2, 4) === Interval(0.25, 1.0)
    @test Interval(1//1, 2//1) / Interval(2//1, 4//1) === Interval(1//4, 1//1)
    @test_throws DivideError Interval(1, 2) / Interval(-1, 1)
    @test_throws DivideError Interval(1, 2) / Interval(0, 1)
    @test_throws DivideError a / 0
    @test a / 2 === Interval(0.5, 1.0)
    @test 2 - a === Interval(0, 1)
    @test Interval(1, 2) + 1//2 === Interval(3//2, 5//2)
    @test Interval(1.0f0, 2.0f0) + 1 isa Interval{Float32}
    @test Interval(1, 2) + big(1.0) isa Interval{BigFloat}
    @test big(1.0) + Interval(1, 2) isa Interval{BigFloat}
    @test Interval(1, 2) + true === Interval(2, 3)

    @test 1.5 in a
    @test !(3 in a)
    @test Interval(1.2, 1.8) in a
    @test !(Interval(0, 1.5) in a)

    @test sprint(show, Interval(1, 2)) == "[1, 2]"
    @test repr(Interval(0.5, 1.0)) == "[0.5, 1.0]"
    @test promote_type(Interval{Int}, Float64) == Interval{Float64}
    @test promote_type(Interval{Int}, Interval{Float32}) == Interval{Float32}
    @test convert(Interval{Float64}, 3) === Interval(3.0, 3.0)
    @test convert(Interval{Float64}, Interval(1, 2)) === Interval(1.0, 2.0)

    x = Interval(1.0, 2.0); y = Interval(0.5, 3.0)
    f(x, y) = (x + y) * (x - y) / Interval(1.0, 2.0) + (-x) * 2.0
    f(x, y)
    @test (@allocated f(x, y)) == 0
    for op in (+, -, *, /)
        op(x, y)
        @test (@allocated op(x, y)) == 0
    end
end
```
