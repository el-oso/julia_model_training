```julia
@enum OpKind::UInt8 ADD MOVE OUT IN JZ JNZ CLEAR

struct Op
    kind::OpKind
    arg::Int        # ADD: amount mod 256; MOVE: offset; JZ/JNZ: index of the matching jump
end

const TAPE_LENGTH = 30_000

# Character position (1-based) of byte index i, for error messages.
charpos(s::String, i::Int) = length(s, 1, i)

"""
    compile_bf(code) -> Vector{Op}

Translate Brainfuck source into instructions: runs of `+`/`-` and `<`/`>` are
folded, `[-]` and `[+]` become `CLEAR`, and each bracket stores the index of
its partner. All other characters are comments. Throws `ArgumentError` for an
unmatched bracket, naming its position.
"""
function compile_bf(code::AbstractString)
    src = String(code)
    cu = codeunits(src)
    n = length(cu)
    ops = Op[]
    open = Tuple{Int,Int}[]          # (op index of '[', byte position)
    i = 1
    while i <= n
        c = cu[i]
        if c == UInt8('+') || c == UInt8('-')
            delta = 0
            while i <= n && (cu[i] == UInt8('+') || cu[i] == UInt8('-'))
                delta += cu[i] == UInt8('+') ? 1 : -1
                i += 1
            end
            d = mod(delta, 256)
            d != 0 && push!(ops, Op(ADD, d))
            continue
        elseif c == UInt8('>') || c == UInt8('<')
            delta = 0
            while i <= n && (cu[i] == UInt8('>') || cu[i] == UInt8('<'))
                delta += cu[i] == UInt8('>') ? 1 : -1
                i += 1
            end
            delta != 0 && push!(ops, Op(MOVE, delta))
            continue
        elseif c == UInt8('[')
            if i + 2 <= n && (cu[i+1] == UInt8('-') || cu[i+1] == UInt8('+')) && cu[i+2] == UInt8(']')
                push!(ops, Op(CLEAR, 0))
                i += 3
                continue
            end
            push!(ops, Op(JZ, 0))            # target patched at the matching ']'
            push!(open, (length(ops), i))
        elseif c == UInt8(']')
            isempty(open) && throw(ArgumentError("unmatched ']' at position $(charpos(src, i))"))
            j, _ = pop!(open)
            push!(ops, Op(JNZ, j))
            ops[j] = Op(JZ, length(ops))
        elseif c == UInt8('.')
            push!(ops, Op(OUT, 0))
        elseif c == UInt8(',')
            push!(ops, Op(IN, 0))
        end
        i += 1
    end
    isempty(open) || throw(ArgumentError("unmatched '[' at position $(charpos(src, last(open)[2]))"))
    return ops
end

"""
    run_bf(code, input="") -> String

Run a Brainfuck program on a 30000-cell tape of `UInt8` (wrapping arithmetic),
reading bytes of `input` with `,` (0 once input is exhausted) and returning the
bytes written by `.` as a `String`. Moving the pointer off either end of the
tape throws an error.
"""
function run_bf(code::AbstractString, input::AbstractString = "")::String
    ops = compile_bf(code)
    tape = zeros(UInt8, TAPE_LENGTH)
    inp = codeunits(String(input))
    ip = 1
    out = UInt8[]
    ptr = 1
    pc = 1
    nops = length(ops)
    while pc <= nops
        op = ops[pc]
        k = op.kind
        if k == ADD
            tape[ptr] += op.arg % UInt8          # UInt8 addition wraps
        elseif k == MOVE
            ptr += op.arg
            1 <= ptr <= TAPE_LENGTH ||
                error("pointer moved off the tape to cell $(ptr - 1) (valid cells: 0:$(TAPE_LENGTH - 1)) at instruction $pc")
        elseif k == JZ
            tape[ptr] == 0 && (pc = op.arg)
        elseif k == JNZ
            tape[ptr] != 0 && (pc = op.arg)
        elseif k == CLEAR
            tape[ptr] = 0x00
        elseif k == OUT
            push!(out, tape[ptr])
        else # IN
            tape[ptr] = ip <= length(inp) ? inp[ip] : 0x00
            ip += 1
        end
        pc += 1
    end
    return String(out)
end
```

### Design

- **Compilation** turns the source into a `Vector{Op}`, where `Op` is a 16-byte `isbits` struct holding an opcode and an `Int` argument:
  - Runs of `+`/`-` are folded into a single `ADD n` (mod 256), and runs of `<`/`>` into a single `MOVE n`. A run that cancels out, such as `+-`, produces no instruction.
  - `[-]` and `[+]` become `CLEAR`.
  - Each `[` (`JZ`) stores the index of its matching `]` (`JNZ`) and vice versa. The interpreter jumps to that index, and the `pc += 1` at the end of the loop body then steps past it. This gives the usual semantics: skip the loop when the cell is 0, and repeat it when the cell is nonzero.
  - All other characters are comments.
- **Errors.**
  - An unmatched `[` or `]` raises an `ArgumentError` that gives the character position. The position counts characters, so it stays correct when Unicode comments come before the bracket.
  - Moving the pointer off the tape raises an error naming the cell (counted from 0) and the instruction. The check runs on every `MOVE`, so the program never reads or writes outside the tape.
- **Tape and I/O.** The tape is 30000 `UInt8` cells, so arithmetic wraps through Julia's ordinary `UInt8` overflow. Output is collected as bytes and converted to a `String` once at the end. `,` reads the next input byte, and once the input is exhausted it stores 0, so `,[.,]` echoes its input and stops.
- **Interpreter loop.** The opcode is an `@enum` backed by `UInt8`, and the loop uses an `if`/`elseif` chain over it, with no dynamic dispatch. Every array access keeps Julia's bounds checks.

### Tests

```julia
using Test
const HELLO = "++++++++[>++++[>++>+++>+++>+<<<<-]>+>+>->>+[<]<-]>>.>---.+++++++..+++.>>.<-.<.+++.------.--------.>>+.>++."
@testset "brainfuck" begin
    @test run_bf(HELLO) == "Hello World!\n"
    @test run_bf(",[.,]", "echo me") == "echo me"
    @test run_bf(",[.,]", "") == ""
    @test run_bf(">,[>,]<[.<]", "abc") == "cba"              # reverse the input
    @test run_bf(",>,<[->+<]>.", "\x03\x04") == "\x07"        # add two input bytes
    @test run_bf("++++[-].") == "\0"
    @test run_bf("-.") == String([0xff])                      # wraparound down
    @test run_bf("-+.") == "\0"                               # folded to nothing
    @test run_bf("+"^256 * ".") == "\0"                       # wraparound up
    @test run_bf("this is + a comment +++ .") == "\x04"
    @test run_bf(">>><<<.") == "\0"
    ops = compile_bf("+++--->><[-]")
    @test ops == [Op(MOVE, 1), Op(CLEAR, 0)]
    @test compile_bf("[[]]") == [Op(JZ, 4), Op(JZ, 3), Op(JNZ, 2), Op(JNZ, 1)]
    @test_throws "unmatched '[' at position 3" run_bf("+.[")
    @test_throws "unmatched ']' at position 4" run_bf("+[]]")
    @test_throws "unmatched '[' at position 2" run_bf("é[")
    @test_throws "pointer moved off the tape to cell -1" run_bf("<")
    @test_throws "pointer moved off the tape to cell 30000" run_bf("+[>+]")
    @test run_bf(">"^29999 * "+.") == "\x01"
end
```

All 19 checks pass. Among them:
- "Hello World!": the classic program from Wikipedia.
- An echo program.
- Programs that read input: one prints the input reversed, one adds two input bytes.
- Clearing, wraparound in both directions, folding, and comments.
- The compiled instruction stream for a few short programs.
- Every error case.

`heavy` runs about 8×10^7 instructions in 163 ms, roughly 2 ns per instruction. Further speedups would come from more peephole patterns, for example compiling `[->+<]` into a single "add this cell to the cell at offset k" instruction.
