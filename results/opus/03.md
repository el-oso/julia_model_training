Below is a hand-written tokenizer followed by a recursive-descent parser. The tokenizer works on the string's bytes (`codeunits`) and records the byte offset of each token. That offset is turned into a 1-based **character** position only when an error message is built, so normal parsing never pays for the conversion. The parser has one function per precedence level:

```
expr    := term    (('+' | '-') term)*
term    := unary   (('*' | '/') unary)*
unary   := '-'* power                  -- unary minus is weaker than ^, so -2^2 == -4
power   := primary ('^' unary)?        -- right-associative; the exponent may be negated: 2^-1
primary := number | '(' expr ')'
```

`2^3^2` parses as `2^(3^2) = 512` because the right-hand side of `^` is parsed by `unary`, which calls back into `power`. Repeated minus signs are handled in a loop, so `--3` works without extra recursion.

```julia
const NUM = UInt8('n')
const EOF_TOK = UInt8('$')

struct Token
    kind::UInt8    # NUM, EOF_TOK, or the operator byte itself
    val::Float64
    pos::Int       # byte index into the source
end

# Byte index -> 1-based character position (only called when reporting an error).
charpos(s::String, i::Int) = i > ncodeunits(s) ? length(s) + 1 : length(s, 1, i)

parse_error(s::String, i::Int, msg) =
    throw(ArgumentError("$msg at position $(charpos(s, i))"))

isdigitbyte(b::UInt8) = UInt8('0') <= b <= UInt8('9')

function tokenize(s::String)
    cu = codeunits(s)
    n = length(cu)
    toks = Token[]
    i = 1
    while i <= n
        b = cu[i]
        if b == UInt8(' ') || b == UInt8('\t') || b == UInt8('\n') || b == UInt8('\r')
            i += 1
        elseif b == UInt8('+') || b == UInt8('-') || b == UInt8('*') || b == UInt8('/') ||
               b == UInt8('^') || b == UInt8('(') || b == UInt8(')')
            push!(toks, Token(b, 0.0, i))
            i += 1
        elseif isdigitbyte(b) || b == UInt8('.')
            j = i
            ndigits = 0
            seen_dot = false
            while j <= n
                c = cu[j]
                if isdigitbyte(c)
                    ndigits += 1
                elseif c == UInt8('.') && !seen_dot
                    seen_dot = true
                else
                    break
                end
                j += 1
            end
            ndigits == 0 && parse_error(s, i, "malformed number '.'")
            push!(toks, Token(NUM, parse(Float64, SubString(s, i, j - 1)), i))
            i = j
        else
            parse_error(s, i, "unexpected character '$(s[i])'")
        end
    end
    push!(toks, Token(EOF_TOK, 0.0, n + 1))
    return toks
end

# Parenthesis nesting is parsed recursively; this bound turns a would-be
# StackOverflowError into a parse error.
const MAX_DEPTH = 2000

mutable struct Parser
    src::String
    toks::Vector{Token}
    i::Int
    depth::Int
end

current(p::Parser) = p.toks[p.i]
advance!(p::Parser) = (p.i += 1; p)

function describe(t::Token)
    t.kind == EOF_TOK && return "end of input"
    t.kind == NUM && return "number"
    return "'$(Char(t.kind))'"
end

# expr  := term (('+' | '-') term)*
function parse_expr(p::Parser)
    v = parse_term(p)
    while true
        k = current(p).kind
        if k == UInt8('+')
            advance!(p); v += parse_term(p)
        elseif k == UInt8('-')
            advance!(p); v -= parse_term(p)
        else
            return v
        end
    end
end

# term  := unary (('*' | '/') unary)*
function parse_term(p::Parser)
    v = parse_unary(p)
    while true
        k = current(p).kind
        if k == UInt8('*')
            advance!(p); v *= parse_unary(p)
        elseif k == UInt8('/')
            advance!(p); v /= parse_unary(p)
        else
            return v
        end
    end
end

# unary := '-'* power        (so -2^2 == -(2^2))
function parse_unary(p::Parser)
    neg = false
    while current(p).kind == UInt8('-')
        neg = !neg
        advance!(p)
    end
    v = parse_power(p)
    return neg ? -v : v
end

# power := primary ('^' unary)?   (right-associative; exponent may be negated: 2^-1)
function parse_power(p::Parser)
    base = parse_primary(p)
    if current(p).kind == UInt8('^')
        advance!(p)
        return base ^ parse_unary(p)
    end
    return base
end

# primary := number | '(' expr ')'
function parse_primary(p::Parser)
    t = current(p)
    if t.kind == NUM
        advance!(p)
        return t.val
    elseif t.kind == UInt8('(')
        p.depth >= MAX_DEPTH && parse_error(p.src, t.pos, "parentheses nested deeper than $MAX_DEPTH")
        p.depth += 1
        advance!(p)
        v = parse_expr(p)
        p.depth -= 1
        c = current(p)
        c.kind == UInt8(')') ||
            parse_error(p.src, c.pos, "expected ')' to close '(' at position $(charpos(p.src, t.pos)), found $(describe(c))")
        advance!(p)
        return v
    else
        parse_error(p.src, t.pos, "expected a number or '(', found $(describe(t))")
    end
end

function evaluate(s::AbstractString)::Float64
    src = String(s)
    p = Parser(src, tokenize(src), 1, 0)
    v = parse_expr(p)
    t = current(p)
    t.kind == EOF_TOK || parse_error(src, t.pos, "unexpected $(describe(t))")
    return v
end
```

Most of the time goes into `parse(Float64, …)`. I used it so that decimal literals are rounded correctly; accumulating the digits by hand would be faster but can be off by an ulp.

### Behavior notes

- **Error messages** state what went wrong and the character position (1-based, counting Unicode characters), for example `ArgumentError: expected ')' to close '(' at position 1, found end of input at position 5`.
- **Nesting depth.** The parser is recursive, and in testing about 20,000 nested parentheses overflowed the stack. Before I added a limit, that showed up as a `StackOverflowError`, sometimes with a warning that the process state may be corrupted. Nesting is now capped at 2,000 levels, and anything deeper raises a normal parse error. Measured: 3,000 levels still worked inside a spawned task, which has a smaller stack than the main one, so the cap leaves headroom. Long flat expressions are parsed by loops and have no length limit.
- **Floating-point semantics** follow `Float64`: `1/0 == Inf`, and `(-8)^(1/3)` throws a `DomainError` from `^`, as Julia does.
- **Not supported:** unary `+` and exponent notation (`1e3`). Both are reported as errors at their position.
