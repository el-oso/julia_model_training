This LRU cache keeps a `Dict` from each key to a slot number and links the slots into a doubly linked list. The links are stored in plain `Int` vectors (`prev`/`next`) rather than as heap-allocated nodes. Because of that, moving an entry to the front only rewrites a few integers, and `get` on a present key allocates nothing. When the cache is full, an insert takes over the slot of the least recently used entry, so the storage never grows beyond `capacity`.

```julia
mutable struct LRU{K,V}
    capacity::Int
    index::Dict{K,Int}   # key => slot
    keys::Vector{K}      # slot => key
    vals::Vector{V}      # slot => value
    prev::Vector{Int}    # slot => more recently used slot (0 = none)
    next::Vector{Int}    # slot => less recently used slot (0 = none)
    head::Int            # most recently used slot (0 when empty)
    tail::Int            # least recently used slot (0 when empty)
end

function LRU{K,V}(capacity::Integer) where {K,V}
    capacity > 0 || throw(ArgumentError("capacity must be positive, got $capacity"))
    index = Dict{K,Int}()
    sizehint!(index, capacity)
    return LRU{K,V}(Int(capacity), index, K[], V[], Int[], Int[], 0, 0)
end

Base.length(c::LRU) = length(c.index)
Base.haskey(c::LRU, key) = haskey(c.index, key)   # does not count as a use

function _unlink!(c::LRU, i::Int)
    p, n = c.prev[i], c.next[i]
    if p == 0
        c.head = n
    else
        c.next[p] = n
    end
    if n == 0
        c.tail = p
    else
        c.prev[n] = p
    end
    return c
end

function _push_front!(c::LRU, i::Int)
    c.prev[i] = 0
    c.next[i] = c.head
    if c.head == 0
        c.tail = i
    else
        c.prev[c.head] = i
    end
    c.head = i
    return c
end

function _touch!(c::LRU, i::Int)
    i == c.head && return c
    _unlink!(c, i)
    return _push_front!(c, i)
end

function Base.get(c::LRU, key, default)
    i = get(c.index, key, 0)
    i == 0 && return default
    _touch!(c, i)
    return c.vals[i]
end

function Base.setindex!(c::LRU{K,V}, value, key) where {K,V}
    k = convert(K, key)
    v = convert(V, value)
    i = get(c.index, k, 0)
    if i != 0                      # update: counts as a use
        c.vals[i] = v
        _touch!(c, i)
    elseif length(c.index) < c.capacity
        push!(c.keys, k)
        push!(c.vals, v)
        push!(c.prev, 0)
        push!(c.next, 0)
        i = length(c.keys)
        c.index[k] = i
        _push_front!(c, i)
    else
        i = c.tail                 # evict the least recently used entry, reuse its slot
        _unlink!(c, i)
        delete!(c.index, c.keys[i])
        c.keys[i] = k
        c.vals[i] = v
        c.index[k] = i
        _push_front!(c, i)
    end
    return c
end
```

Design decisions:
- **Every operation is O(1).** Each one does one or two `Dict` operations (expected O(1)) plus a fixed number of link updates.
- **`get` does not allocate.** It does a `Dict` lookup that returns an `Int` slot (default `0`) and then updates a few integers. Nothing is boxed as long as `K` and `V` are concrete. The test below checks this with `@allocated`, for both `Int` and `String` keys.
- **What counts as a use.** `get` and `setindex!` (both inserting and updating) count as uses. `haskey` does not, so you can check for a key without changing the eviction order.
- **Conversion.** `setindex!` converts the key and value to `K`/`V`, so `c[1] = 2` works on an `LRU{Int,Float64}`.

### Tests

```julia
using Test

@testset "LRU" begin
    c = LRU{Int,String}(2)
    @test length(c) == 0
    @test get(c, 1, "none") == "none"
    c[1] = "a"; c[2] = "b"
    @test length(c) == 2
    @test get(c, 1, "") == "a"      # 1 becomes most recent
    c[3] = "c"                      # evicts 2
    @test !haskey(c, 2)
    @test haskey(c, 1) && haskey(c, 3)
    c[1] = "A"                      # update counts as a use
    c[4] = "d"                      # evicts 3
    @test !haskey(c, 3)
    @test get(c, 1, "") == "A"
    @test length(c) == 2
    @test_throws ArgumentError LRU{Int,Int}(0)

    c1 = LRU{Symbol,Int}(1)
    c1[:x] = 1; c1[:y] = 2
    @test !haskey(c1, :x) && get(c1, :y, 0) == 2

    # randomized comparison against a simple reference model
    for cap in (1, 3, 10)
        c = LRU{Int,Int}(cap)
        order = Int[]; ref = Dict{Int,Int}()       # order: most recent last
        for _ in 1:5000
            k = rand(1:2cap+1)
            if rand(Bool)
                v = rand(Int)
                c[k] = v
                if haskey(ref, k)
                    deleteat!(order, findfirst(==(k), order))
                elseif length(order) == cap
                    delete!(ref, popfirst!(order))
                end
                push!(order, k); ref[k] = v
            else
                @test get(c, k, nothing) == get(ref, k, nothing)
                if haskey(ref, k)
                    deleteat!(order, findfirst(==(k), order)); push!(order, k)
                end
            end
            @test length(c) == length(ref)
        end
    end

    # get on a present key does not allocate
    c = LRU{Int,Float64}(100)
    for k in 1:100; c[k] = k; end
    f(c) = get(c, 50, 0.0) + get(c, 7, 0.0)
    f(c)
    @test (@allocated f(c)) == 0
    cs = LRU{String,Int}(10)
    cs["hello"] = 1; cs["world"] = 2
    g(c, k) = get(c, k, 0)
    key = "hello"; g(cs, key)
    @test (@allocated g(cs, key)) == 0
end
```
