Design: a `Dict{K,Int}` maps each key to a slot number; slot data lives in parallel arrays that form a doubly linked list (indices instead of pointers; 0 means "none"). `head` is the most recently used slot, `tail` the least recently used. Every operation is a hash lookup plus a constant number of array writes. `get` on a hit does a lookup and relinks integers only, so it does not allocate.

```julia
mutable struct LRU{K,V}
    index::Dict{K,Int}
    keys::Vector{K}
    vals::Vector{V}
    prev::Vector{Int}
    next::Vector{Int}
    head::Int          # most recently used slot, 0 if empty
    tail::Int          # least recently used slot, 0 if empty
    capacity::Int

    function LRU{K,V}(capacity::Integer) where {K,V}
        capacity > 0 || throw(ArgumentError("capacity must be positive, got $capacity"))
        c = Int(capacity)
        new{K,V}(Dict{K,Int}(), sizehint!(K[], c), sizehint!(V[], c),
                 sizehint!(Int[], c), sizehint!(Int[], c), 0, 0, c)
    end
end

Base.length(c::LRU) = length(c.index)
Base.haskey(c::LRU, key) = haskey(c.index, key)   # does not count as a use

function unlink!(c::LRU, i::Int)
    p, n = c.prev[i], c.next[i]
    p == 0 ? (c.head = n) : (c.next[p] = n)
    n == 0 ? (c.tail = p) : (c.prev[n] = p)
    return nothing
end

function pushfront!(c::LRU, i::Int)
    c.prev[i] = 0
    c.next[i] = c.head
    c.head != 0 && (c.prev[c.head] = i)
    c.head = i
    c.tail == 0 && (c.tail = i)
    return nothing
end

function touch!(c::LRU, i::Int)
    if c.head != i
        unlink!(c, i)
        pushfront!(c, i)
    end
    return nothing
end

function Base.get(c::LRU, key, default)
    i = get(c.index, key, 0)
    i == 0 && return default
    touch!(c, i)
    return c.vals[i]
end

function Base.setindex!(c::LRU{K,V}, value, key) where {K,V}
    k = convert(K, key)
    v = convert(V, value)
    i = get(c.index, k, 0)
    if i != 0                       # update and mark as used
        c.vals[i] = v
        touch!(c, i)
    elseif length(c.index) < c.capacity   # room left: allocate a new slot
        push!(c.keys, k); push!(c.vals, v)
        push!(c.prev, 0); push!(c.next, 0)
        i = length(c.keys)
        c.index[k] = i
        pushfront!(c, i)
    else                            # full: recycle the LRU slot
        i = c.tail
        delete!(c.index, c.keys[i])
        unlink!(c, i)
        c.keys[i] = k
        c.vals[i] = v
        c.index[k] = i
        pushfront!(c, i)
    end
    return c
end
```

Behavior notes:

- `get` counts as a use; `haskey` and `length` do not (a membership test that reordered the cache would be surprising). Say so if you want `haskey` to count.
- Slots are only ever added until the cache is full and are then recycled, so `length(c.keys) == length(c.index)` always holds (there is no `delete!`).
- Updating an existing key also counts as a use and never evicts.
- `get(c, key, default)` returns `Union{V, typeof(default)}`; with a `default` of type `V` it is concretely typed.

## Tests

```julia
using Test

@testset "LRU" begin
    c = LRU{Int,String}(2)
    @test length(c) == 0
    c[1] = "a"; c[2] = "b"
    @test length(c) == 2
    @test get(c, 1, "") == "a"          # 1 is now most recent
    c[3] = "c"                          # evicts 2
    @test !haskey(c, 2)
    @test haskey(c, 1) && haskey(c, 3)
    @test get(c, 2, "none") == "none"

    c[1] = "A"                          # update, no eviction, 1 most recent
    c[4] = "d"                          # evicts 3
    @test !haskey(c, 3)
    @test get(c, 1, "") == "A"
    @test length(c) == 2

    @test_throws ArgumentError LRU{Int,Int}(0)

    # capacity 1
    d = LRU{String,Int}(1)
    d["x"] = 1; d["y"] = 2
    @test !haskey(d, "x") && get(d, "y", 0) == 2

    # model-based random test against a naive list
    cap = 5
    lru = LRU{Int,Int}(cap)
    model = Pair{Int,Int}[]            # front = most recent
    for _ in 1:10_000
        k = rand(1:12)
        if rand() < 0.5
            v = rand(Int)
            lru[k] = v
            filter!(p -> p.first != k, model)
            pushfirst!(model, k => v)
            length(model) > cap && pop!(model)
        else
            got = get(lru, k, nothing)
            idx = findfirst(p -> p.first == k, model)
            if idx === nothing
                @test got === nothing
            else
                @test got == model[idx].second
                pushfirst!(model, popat!(model, idx))
            end
        end
        @test length(lru) == length(model)
    end
end

@testset "get does not allocate on a hit" begin
    c = LRU{Int,Int}(100)
    for i in 1:100; c[i] = i; end
    probe(c) = (get(c, 50, 0); get(c, 7, 0); @allocated(get(c, 50, 0)))
    probe(c)                      # warm up
    @test probe(c) == 0
end
```

All tests above passed when run (15078 checks, including the model-based random test and the zero-allocation check for `get` on a hit).
