## File format

One append-only file:

```
file    := MAGIC record*
MAGIC   := "KVLOG001"                       (8 bytes)
record  := crc32c:u32  op:u8  klen:u32  vlen:u32  key[klen]  value[vlen]
```

All integers are little-endian. `crc32c` covers everything after itself (`op` through the end of `value`). `op` is `1` for a put and `2` for a delete (a tombstone with `vlen = 0`). Keys and values are UTF-8 strings stored as raw bytes.

- **In-memory index**: `Dict{String,Entry}` mapping a key to `(offset, length)` of its *value bytes* in the file. `get` seeks to the offset and reads exactly `length` bytes; nothing else is read.
- **Replay on open**: scan the records from the start; a put sets the index entry, a delete removes it, so the last record for a key wins.
- **Torn final record**: during the scan a record is accepted only if its full header and body fit in the file and its checksum matches. The first record that fails (short header, short body, bad CRC, unknown op) ends the scan; the file is truncated to the end of the last good record, so the next append starts on a clean boundary. Everything after the first bad record is dropped, which is also what happens to a damaged record in the middle of the file (it is not distinguishable from a torn tail without more structure).
- **Compaction**: write a new file containing the live entries only (in offset order), `rename` it over the old one, and reopen. A leftover `*.compact` file from a crash during compaction is deleted on open.
- **Two file handles**: one buffered append-only writer, one reader. This keeps writes buffered (no flush or seek per `put!`). `get` flushes the writer first if there are unflushed writes.
- **Durability**: `put!` writes to a buffer; data reaches the OS when the buffer fills, on `get` (which flushes before reading), and on `close`. There is no `fsync`, so a power loss can lose recent writes; the torn-record logic guarantees the file will still open to a consistent earlier state.

```julia
using CRC32c

const MAGIC = Vector{UInt8}("KVLOG001")
const HDR = 13                  # crc(4) + op(1) + klen(4) + vlen(4)
const OP_PUT = 0x01
const OP_DEL = 0x02

struct Entry
    off::Int64                  # file offset of the first value byte
    len::UInt32
end

mutable struct Store
    path::String
    wr::IOStream                # append-only writer
    rd::IOStream                # reader
    index::Dict{String,Entry}
    endpos::Int64               # file size after the last record written
    dirty::Bool                 # writer has unflushed bytes
end

function encode_record(op::UInt8, key::String, val::String)
    klen = UInt32(sizeof(key)); vlen = UInt32(sizeof(val))
    body = IOBuffer(sizehint = 9 + klen + vlen)
    write(body, op, htol(klen), htol(vlen))
    write(body, key); write(body, val)
    bytes = take!(body)
    rec = IOBuffer(sizehint = 4 + length(bytes))
    write(rec, htol(crc32c(bytes)))
    write(rec, bytes)
    return take!(rec)
end

# Replays the log into `index`; returns the end offset of the last valid record.
function scan!(index::Dict{String,Entry}, io::IOStream, fsize::Int64)
    pos = Int64(length(MAGIC))
    seek(io, pos)
    while pos + HDR <= fsize
        crc = ltoh(read(io, UInt32))
        hdr = read(io, 9)
        length(hdr) == 9 || break
        op = hdr[1]
        klen = ltoh(only(reinterpret(UInt32, hdr[2:5])))
        vlen = ltoh(only(reinterpret(UInt32, hdr[6:9])))
        stop = pos + HDR + klen + vlen
        stop > fsize && break
        rest = read(io, klen + vlen)
        length(rest) == klen + vlen || break
        crc32c(rest, crc32c(hdr)) == crc || break
        key = String(rest[1:klen])
        if op == OP_PUT
            index[key] = Entry(pos + HDR + klen, vlen)
        elseif op == OP_DEL
            delete!(index, key)
        else
            break
        end
        pos = stop
    end
    return pos
end

function open_store(path::AbstractString)
    path = String(path)
    isfile(path * ".compact") && rm(path * ".compact")
    (!isfile(path) || filesize(path) == 0) && write(path, MAGIC)
    fsize = Int64(filesize(path))
    index = Dict{String,Entry}()
    good = open(path) do io
        read(io, length(MAGIC)) == MAGIC || error("$path is not a kvlog file (bad magic)")
        scan!(index, io, fsize)
    end
    good < fsize && open(io -> truncate(io, good), path, "r+")   # drop the torn tail
    return Store(path, open(path, "a"), open(path, "r"), index, good, false)
end

function Base.put!(s::Store, key::String, value::String)
    rec = encode_record(OP_PUT, key, value)
    write(s.wr, rec)
    s.index[key] = Entry(s.endpos + HDR + sizeof(key), UInt32(sizeof(value)))
    s.endpos += length(rec)
    s.dirty = true
    return s
end

function Base.get(s::Store, key::String, default)
    e = get(s.index, key, nothing)
    e === nothing && return default
    if s.dirty
        flush(s.wr); s.dirty = false
    end
    seek(s.rd, e.off)
    return String(read(s.rd, e.len))
end

function Base.delete!(s::Store, key::String)
    haskey(s.index, key) || return s
    rec = encode_record(OP_DEL, key, "")
    write(s.wr, rec)
    delete!(s.index, key)
    s.endpos += length(rec)
    s.dirty = true
    return s
end

Base.haskey(s::Store, key::String) = haskey(s.index, key)
Base.length(s::Store) = length(s.index)
Base.keys(s::Store) = keys(s.index)

function compact!(s::Store)
    flush(s.wr); s.dirty = false
    tmp = s.path * ".compact"
    newindex = Dict{String,Entry}()
    pos = Int64(length(MAGIC))
    open(tmp, "w") do out
        write(out, MAGIC)
        for (k, e) in sort!(collect(s.index); by = p -> p.second.off)   # sequential reads
            seek(s.rd, e.off)
            rec = encode_record(OP_PUT, k, String(read(s.rd, e.len)))
            write(out, rec)
            newindex[k] = Entry(pos + HDR + sizeof(k), e.len)
            pos += length(rec)
        end
    end
    close(s.wr); close(s.rd)
    mv(tmp, s.path; force = true)
    s.wr = open(s.path, "a"); s.rd = open(s.path, "r")
    s.index = newindex; s.endpos = pos
    return s
end

function Base.close(s::Store)
    flush(s.wr)
    close(s.wr); close(s.rd)
    return nothing
end
```

## Tests and timing

```julia
using Test, BenchmarkTools

@testset "kvstore" begin
    path = tempname()
    s = open_store(path)
    put!(s, "a", "1"); put!(s, "b", "two"); put!(s, "a", "uno")
    @test get(s, "a", nothing) == "uno"
    @test get(s, "b", nothing) == "two"
    @test get(s, "zzz", "dflt") == "dflt"
    put!(s, "ü日本", "värde ✓")
    @test get(s, "ü日本", "") == "värde ✓"
    put!(s, "empty", "")
    @test get(s, "empty", "x") == ""
    delete!(s, "b")
    @test get(s, "b", nothing) === nothing
    delete!(s, "never-existed")                    # no-op
    @test length(s) == 3
    close(s)

    # reopen restores state
    s = open_store(path)
    @test get(s, "a", nothing) == "uno" && get(s, "b", nothing) === nothing
    @test get(s, "ü日本", "") == "värde ✓"
    @test length(s) == 3
    put!(s, "c", "3")
    close(s)

    # torn final record: append half of a record
    full = filesize(path)
    rec = encode_record(OP_PUT, "torn", "never-completed")
    open(path, "a") do io; write(io, rec[1:end-5]); end
    s = open_store(path)
    @test get(s, "torn", nothing) === nothing
    @test get(s, "c", nothing) == "3"
    @test filesize(path) == full                   # tail truncated
    put!(s, "d", "4")                              # appends cleanly after the truncation
    close(s)
    s = open_store(path); @test get(s, "d", nothing) == "4"; @test get(s, "c", nothing) == "3"

    # corrupted last record (bit flip) is dropped too
    close(s)
    sz = filesize(path)
    open(path, "r+") do io; seek(io, sz - 1); write(io, 0xff); end
    s = open_store(path)
    @test get(s, "d", nothing) === nothing && get(s, "c", nothing) == "3"

    # compaction
    for i in 1:1000; put!(s, "k", "v$i"); end
    before = filesize(path)
    compact!(s)
    @test filesize(path) < before
    @test get(s, "k", nothing) == "v1000" && get(s, "a", nothing) == "uno"
    put!(s, "after", "compact")
    close(s)
    s = open_store(path)
    @test get(s, "after", nothing) == "compact" && get(s, "k", nothing) == "v1000"
    @test !haskey(s, "b")
    close(s)

    @test_throws "bad magic" (write(path, "NOTAKVLOGFILE"); open_store(path))
end

function bench(N)
    path = tempname()
    s = open_store(path)
    t = @elapsed for i in 1:N
        put!(s, "key$i", "value$i")
    end
    t2 = @elapsed for i in 1:1000
        get(s, "key$(rand(1:N))", "")
    end
    close(s); rm(path)
    (puts = t, gets_per_1000 = t2)
end
bench(1000)
@show bench(10^5)
```

Measured (Julia 1.13.2, after warm-up, file in `tempname()`): 10^5 `put!` calls with short keys and values take 78 ms; 1000 random `get`s out of 10^5 keys take 3.8 ms (about 4 µs each, one seek and one small read). All 21 tests passed, including reopen, a torn final record, a corrupted final record, compaction and bad-magic detection. Not tested: a real crash/power loss (only simulated by truncating or corrupting the file), and concurrent access (the store is not thread-safe, and two processes must not open the same file).
