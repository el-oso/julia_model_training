Runs are compared with `isequal`, so `NaN` runs together, `missing` runs together, and `0.0`/`-0.0` stay separate (they are distinguishable values). The result element type is `eltype(v)`, so both functions are type-stable.

```julia
function rle(v::AbstractVector)
    values = eltype(v)[]
    counts = Int[]
    isempty(v) && return values, counts
    cur = v[firstindex(v)]
    n = 1
    for i in firstindex(v)+1:lastindex(v)
        x = v[i]
        if isequal(x, cur)
            n += 1
        else
            push!(values, cur)
            push!(counts, n)
            cur = x
            n = 1
        end
    end
    push!(values, cur)
    push!(counts, n)
    return values, counts
end

function unrle(values::AbstractVector, counts::AbstractVector{<:Integer})
    axes(values) == axes(counts) ||
        throw(DimensionMismatch("values and counts must have the same axes: $(axes(values)) vs $(axes(counts))"))
    all(>=(0), counts) || throw(ArgumentError("counts must be non-negative"))
    out = Vector{eltype(values)}(undef, sum(counts; init=0))
    k = 0
    for i in eachindex(values, counts)
        c = counts[i]
        fill!(view(out, k+1:k+c), values[i])
        k += c
    end
    return out
end
```

Notes:

- Indexing uses `firstindex`/`lastindex`/`eachindex`, so `OffsetArray`s and views work. The outputs are ordinary 1-based `Vector`s; `unrle` returns a 1-based `Vector`, so for an offset input `unrle(rle(v)...) == v` holds elementwise (`collect(v)`), not with matching axes.
- `isequal` rather than `==` is deliberate: `NaN == NaN` is `false` (every NaN would be its own run) and `missing == x` is `missing` (which would throw in an `if`).
- Output is built with `push!` (amortized O(1)); a two-pass version that counts runs first and then fills preallocated arrays is an option if allocation growth matters.

## Tests

```julia
using Test
# OffsetArrays is the only non-Base package, and only used in the tests.
using OffsetArrays

@testset "rle/unrle" begin
    @test rle(Int[]) == (Int[], Int[])
    @test unrle(Int[], Int[]) == Int[]
    @test rle([1,1,2,3,3,3]) == ([1,2,3], [2,1,3])
    @test unrle([1,2,3], [2,1,3]) == [1,1,2,3,3,3]

    # NaN runs together; 0.0 and -0.0 do not
    v, c = rle([NaN, NaN, 1.0, 0.0, -0.0])
    @test c == [2, 1, 1, 1]
    @test isnan(v[1])

    # missing
    x = [missing, missing, 1, 1, missing]
    v, c = rle(x)
    @test c == [2, 2, 1]
    @test isequal(unrle(v, c), x)
    @test eltype(v) == Union{Missing,Int}

    # views and ranges
    a = collect(1:10)
    @test rle(view(a, 2:4)) == ([2,3,4], [1,1,1])
    @test rle(1:3) == ([1,2,3], [1,1,1])

    # OffsetArrays
    o = OffsetArray([5,5,6], -3:-1)
    @test rle(o) == ([5,6], [2,1])

    # strings, non-isbits
    @test rle(["a","a","b"]) == (["a","b"], [2,1])

    # round trip, random
    for _ in 1:100
        r = rand(1:3, rand(0:50))
        @test unrle(rle(r)...) == r
    end

    # type stability
    @test @inferred(rle([1,1,2])) isa Tuple{Vector{Int},Vector{Int}}
    @inferred unrle([1,2], [2,3])

    @test_throws ArgumentError unrle([1,2], [1,-1])
    @test_throws DimensionMismatch unrle([1,2], [1])
end
```

## Benchmark (10^7 elements)

Measured on Julia 1.13.2 (single thread, BenchmarkTools minimum): `rle` 9.4 ms (58 allocations, 33.7 MiB), `unrle` 45.9 ms (3 allocations, 76 MiB, which is the output vector).
