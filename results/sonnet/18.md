```julia
using LinearAlgebra

"""
    mymul!(C, A, B; bs = 64) -> C

`C = A * B` for dense matrices of any numeric element type, with cache
blocking and column-major loops (no BLAS). `C` must not alias `A` or `B`.
"""
function mymul!(C::AbstractMatrix, A::AbstractMatrix, B::AbstractMatrix; bs::Integer = 128)
    Base.require_one_based_indexing(C, A, B)
    m, l = size(A)
    l2, n = size(B)
    (l == l2 && size(C) == (m, n)) ||
        throw(DimensionMismatch("C is $(size(C)), A is $(size(A)), B is $(size(B))"))
    (C === A || C === B) && throw(ArgumentError("C must not alias A or B"))
    bs >= 1 || throw(ArgumentError("block size must be positive"))
    fill!(C, zero(eltype(C)))
    for jj in 1:bs:n, kk in 1:bs:l, ii in 1:bs:m
        for j in jj:min(jj + bs - 1, n)
            for k in kk:min(kk + bs - 1, l)
                b = B[k, j]
                # i, k, j are inside 1:m, 1:l, 1:n by the loop bounds and the dimension
                # check above, so the bounds checks can go; they block vectorization.
                @inbounds @simd for i in ii:min(ii + bs - 1, m)   # stride 1 in A and C
                    C[i, j] += A[i, k] * b
                end
            end
        end
    end
    return C
end
```

The three block loops go over blocks of `C` columns (`jj`), the inner dimension (`kk`) and rows (`ii`); within a block the loop nest is `j, k, i`, so the innermost loop walks down a column of `A` and of `C` (stride 1). The `bs x bs` blocks of `A` are reused across the `bs` columns of the block, which is the cache reuse. `C` is zeroed first and accumulated into, so the result has `mul!` semantics (`C` overwritten, not `C += A*B`).

## Tests and benchmark

```julia
using Test, BenchmarkTools, Random

@testset "mymul!" begin
    rng = Xoshiro(1)
    for (m, l, n) in ((1, 1, 1), (5, 7, 3), (64, 64, 64), (65, 130, 67), (100, 3, 200))
        A = randn(rng, m, l); B = randn(rng, l, n); C = zeros(m, n)
        @test mymul!(C, A, B) ≈ A * B
    end
    # Int: exact
    A = rand(rng, -9:9, 70, 80); B = rand(rng, -9:9, 80, 90); C = zeros(Int, 70, 90)
    @test mymul!(C, A, B) == A * B
    # Complex{Float32}
    A = rand(rng, ComplexF32, 33, 40); B = rand(rng, ComplexF32, 40, 21); C = zeros(ComplexF32, 33, 21)
    @test mymul!(C, A, B) ≈ A * B
    @test eltype(C) == ComplexF32
    # existing C contents are overwritten
    C = fill(99.0, 2, 2); mymul!(C, [1.0 0; 0 1], [1.0 2; 3 4]); @test C == [1.0 2; 3 4]
    @test_throws DimensionMismatch mymul!(zeros(2, 2), zeros(2, 3), zeros(2, 2))
    M = zeros(3, 3); @test_throws ArgumentError mymul!(M, M, ones(3, 3))
end

n = 512
A = randn(Xoshiro(1), n, n); B = randn(Xoshiro(2), n, n); C = similar(A)
mymul!(C, A, B); mul!(C, A, B)
@btime mymul!($C, $A, $B);
@btime mul!($C, $A, $B);
```

## Measured (Julia 1.13.2, n = 512, `Float64`, BenchmarkTools)

| | time |
|---|---|
| `mymul!` (this code, block 128) | 13.3 ms |
| `mul!` (OpenBLAS), 1 thread | 5.4 ms (`OPENBLAS_NUM_THREADS=1`) |
| `mul!` (OpenBLAS), default threads on 6 cores | 2.06 ms |
| `mymul!` without `@inbounds @simd` | about 80-86 ms |

So `mymul!` is about 2.5x slower than single-threaded OpenBLAS and about 6.5x slower than the threaded default. The `@inbounds @simd` on the innermost loop is what makes it competitive: without it the same code took 80-86 ms (measured with block sizes 32 to 256), so here the bounds checks were what blocked vectorization. The annotation is safe because `i`, `k`, `j` come from `min`-clamped ranges over the validated dimensions; the tests also pass under `--check-bounds=yes`. Block sizes tried with the annotation: 32: 24.8 ms, 64: 17.1 ms, 128: 13.7 ms, 256: 18.0 ms, hence the default of 128. An additional 4-column register-blocking variant gave 12.7 ms, a small enough gain that I did not keep the extra code. The tests cover several shapes including non-multiples of the block size, exact `Int`, `Complex{Float32}`, overwrite semantics, dimension mismatch and aliasing; all passed.
