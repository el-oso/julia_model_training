Grammar (lowest to highest precedence):

```
expr   := term (('+' | '-') term)*
term   := unary (('*' | '/') unary)*
unary  := '-' unary | power
power  := atom ('^' unary)?        # right-associative; exponent may be negated
atom   := NUMBER | '(' expr ')'
```

Because `unary` wraps `power`, `-2^2 == -(2^2) == -4`; because the exponent is a `unary`, `2^-1 == 0.5` and `2^3^2 == 2^(3^2) == 512`.

The tokenizer is a loop over `codeunits` (all operators are ASCII) that fills three parallel vectors; positions are 1-based byte indices into the string (identical to character positions for ASCII input).

```julia
struct ExprError <: Exception
    msg::String
    pos::Int
end
Base.showerror(io::IO, e::ExprError) = print(io, "ExprError: ", e.msg, " at position ", e.pos)

@enum Tok::UInt8 NUM PLUS MINUS STAR SLASH CARET LPAREN RPAREN EOF

const TOKTEXT = Dict(NUM => "number", PLUS => "'+'", MINUS => "'-'", STAR => "'*'",
                     SLASH => "'/'", CARET => "'^'", LPAREN => "'('", RPAREN => "')'",
                     EOF => "end of input")

struct Tokens
    kind::Vector{Tok}
    val::Vector{Float64}
    pos::Vector{Int}
end

isdigitb(c::UInt8) = UInt8('0') <= c <= UInt8('9')

function tokenize(s::AbstractString)
    cu = codeunits(s)
    n = length(cu)
    t = Tokens(Tok[], Float64[], Int[])
    sizehint!(t.kind, n ÷ 2); sizehint!(t.val, n ÷ 2); sizehint!(t.pos, n ÷ 2)
    emit!(k, x, p) = (push!(t.kind, k); push!(t.val, x); push!(t.pos, p))
    i = 1
    while i <= n
        c = cu[i]
        if c == UInt8(' ') || c == UInt8('\t') || c == UInt8('\n') || c == UInt8('\r')
            i += 1
        elseif isdigitb(c) || c == UInt8('.')
            start = i
            ndigits = 0
            seendot = false
            while i <= n
                d = cu[i]
                if isdigitb(d)
                    ndigits += 1
                elseif d == UInt8('.')
                    seendot && throw(ExprError("second decimal point in number", i))
                    seendot = true
                else
                    break
                end
                i += 1
            end
            ndigits == 0 && throw(ExprError("a decimal point must be next to a digit", start))
            # the span contains only ASCII digits and '.', so byte indices are valid string indices
            emit!(NUM, parse(Float64, SubString(s, start, i - 1)), start)
        else
            k = c == UInt8('+') ? PLUS  :
                c == UInt8('-') ? MINUS :
                c == UInt8('*') ? STAR  :
                c == UInt8('/') ? SLASH :
                c == UInt8('^') ? CARET :
                c == UInt8('(') ? LPAREN :
                c == UInt8(')') ? RPAREN :
                throw(ExprError("unexpected character '$(s[thisind(s, i)])'", i))
            emit!(k, 0.0, i)
            i += 1
        end
    end
    emit!(EOF, 0.0, n + 1)
    return t
end

mutable struct Parser
    t::Tokens
    i::Int
end

kind(p::Parser) = @inbounds p.t.kind[p.i]

function parse_atom(p::Parser)
    k = kind(p)
    if k == NUM
        x = p.t.val[p.i]
        p.i += 1
        return x
    elseif k == LPAREN
        open = p.t.pos[p.i]
        p.i += 1
        x = parse_expr(p)
        kind(p) == RPAREN ||
            throw(ExprError("expected ')' to close '(' opened at position $open, found $(TOKTEXT[kind(p)])", p.t.pos[p.i]))
        p.i += 1
        return x
    else
        throw(ExprError("expected a number or '(', found $(TOKTEXT[k])", p.t.pos[p.i]))
    end
end

function parse_unary(p::Parser)
    neg = false
    while kind(p) == MINUS      # iterative: long runs of '-' do not deepen the stack
        neg = !neg
        p.i += 1
    end
    x = parse_power(p)
    return neg ? -x : x
end

function parse_power(p::Parser)
    base = parse_atom(p)
    if kind(p) == CARET
        p.i += 1
        return base^parse_unary(p)
    end
    return base
end

function parse_term(p::Parser)
    x = parse_unary(p)
    while true
        k = kind(p)
        if k == STAR
            p.i += 1; x *= parse_unary(p)
        elseif k == SLASH
            p.i += 1; x /= parse_unary(p)
        else
            return x
        end
    end
end

function parse_expr(p::Parser)
    x = parse_term(p)
    while true
        k = kind(p)
        if k == PLUS
            p.i += 1; x += parse_term(p)
        elseif k == MINUS
            p.i += 1; x -= parse_term(p)
        else
            return x
        end
    end
end

function evaluate(s::AbstractString)::Float64
    p = Parser(tokenize(s), 1)
    x = parse_expr(p)
    kind(p) == EOF ||
        throw(ExprError("unexpected $(TOKTEXT[kind(p)])", p.t.pos[p.i]))
    return x
end
```

The `@inbounds` in `kind` is sound because the token vector always ends with `EOF` and the parser never advances past it (every `p.i += 1` follows a check that the current token is not `EOF`). Remove it if you prefer; it is not what makes this fast.

## Behavior and limits

- Division by zero follows IEEE (`1/0 == Inf`); `(-8)^0.5` throws Julia's `DomainError` from `^`.
- Recursion depth grows with nesting of parentheses and with `^` chains (`2^2^2^...`); `+ - * /` chains and runs of unary minus are iterative. A 10^5-character expression made of 50000 nested parentheses would recurse 50000 deep; I did not test where the stack gives out, so use a shunting-yard variant if you need arbitrary nesting.
- No exponent notation (`1e5`), no unary plus (not in the spec).
