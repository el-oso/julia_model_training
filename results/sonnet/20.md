Compilation turns the program into a vector of isbits instructions `(op, arg, lo, hi)`:

- runs of `+`/`-` become one `ADD` with the net amount mod 256; a net of zero emits nothing;
- runs of `>`/`<` become one `MOVE` with the net shift, plus the lowest and highest offset reached inside the run, so the tape-bounds check is exact even though the run is folded;
- `[` / `]` become `JZ` / `JNZ` with the matching index precomputed (a stack at compile time);
- `[-]` (and `[+]`) becomes `CLEAR`;
- anything that is not one of `+-<>[].,` is a comment and is skipped.

```julia
struct BFError <: Exception
    msg::String
end
Base.showerror(io::IO, e::BFError) = print(io, "BFError: ", e.msg)

const TAPE_SIZE = 30000

const ADD, MOVE, OUT, IN, JZ, JNZ, CLEAR = 0x01, 0x02, 0x03, 0x04, 0x05, 0x06, 0x07

struct Instr
    op::UInt8
    arg::Int        # ADD: amount mod 256; MOVE: net shift; JZ/JNZ: index of the partner
    lo::Int         # MOVE: smallest offset reached inside the run (<= 0)
    hi::Int         # MOVE: largest offset reached inside the run (>= 0)
end
Instr(op, arg = 0) = Instr(op, arg, 0, 0)

iscmd(c::UInt8) = c in b"+-<>[].,"

function compile(code::AbstractString)
    cu = codeunits(code)
    n = length(cu)
    prog = Instr[]
    stack = Int[]               # indices of pending JZ instructions
    stackpos = Int[]            # their source positions, for error messages
    i = 1
    while i <= n
        c = cu[i]
        if c == UInt8('+') || c == UInt8('-')
            net = 0
            while i <= n && (!iscmd(cu[i]) || cu[i] == UInt8('+') || cu[i] == UInt8('-'))
                cu[i] == UInt8('+') && (net += 1)
                cu[i] == UInt8('-') && (net -= 1)
                i += 1
            end
            net = mod(net, 256)
            net != 0 && push!(prog, Instr(ADD, net))
        elseif c == UInt8('>') || c == UInt8('<')
            cur = lo = hi = 0
            while i <= n && (!iscmd(cu[i]) || cu[i] == UInt8('>') || cu[i] == UInt8('<'))
                cu[i] == UInt8('>') && (cur += 1)
                cu[i] == UInt8('<') && (cur -= 1)
                lo = min(lo, cur); hi = max(hi, cur)
                i += 1
            end
            cur != 0 || lo != 0 || hi != 0 ? push!(prog, Instr(MOVE, cur, lo, hi)) : nothing
        elseif c == UInt8('.')
            push!(prog, Instr(OUT)); i += 1
        elseif c == UInt8(',')
            push!(prog, Instr(IN)); i += 1
        elseif c == UInt8('[')
            push!(prog, Instr(JZ)); push!(stack, length(prog)); push!(stackpos, i); i += 1
        elseif c == UInt8(']')
            isempty(stack) && throw(BFError("unmatched ']' at position $i"))
            j = pop!(stack); pop!(stackpos)
            if length(prog) == j + 1 && prog[end].op == ADD && (prog[end].arg == 255 || prog[end].arg == 1)
                resize!(prog, j - 1)            # `[-]` / `[+]`: drop the loop and the ADD
                push!(prog, Instr(CLEAR))
            else
                push!(prog, Instr(JNZ, j))
                prog[j] = Instr(JZ, length(prog))
            end
            i += 1
        else
            i += 1                              # comment character
        end
    end
    isempty(stack) || throw(BFError("unmatched '[' at position $(stackpos[1])"))
    return prog
end

function run_bf(code::AbstractString, input::AbstractString = "")::String
    prog = compile(code)
    tape = zeros(UInt8, TAPE_SIZE)
    inp = codeunits(input)
    inpos = 1
    out = UInt8[]
    ptr = 1
    pc = 1
    n = length(prog)
    while pc <= n
        ins = prog[pc]
        op = ins.op
        if op == ADD
            tape[ptr] += UInt8(ins.arg)                 # UInt8 arithmetic wraps
        elseif op == MOVE
            if ptr + ins.lo < 1
                throw(BFError("pointer left the tape: moved to cell $(ptr + ins.lo), valid cells are 1:$TAPE_SIZE"))
            elseif ptr + ins.hi > TAPE_SIZE
                throw(BFError("pointer left the tape: moved to cell $(ptr + ins.hi), valid cells are 1:$TAPE_SIZE"))
            end
            ptr += ins.arg
        elseif op == OUT
            push!(out, tape[ptr])
        elseif op == IN
            tape[ptr] = inpos <= length(inp) ? inp[inpos] : 0x00     # EOF reads as 0
            inpos += 1
        elseif op == JZ
            if tape[ptr] == 0
                pc = ins.arg
            end
        elseif op == JNZ
            if tape[ptr] != 0
                pc = ins.arg
            end
        else # CLEAR
            tape[ptr] = 0x00
        end
        pc += 1
    end
    return String(out)
end
```

Behavior notes:

- Output is the raw bytes written by `.`; if the program writes bytes that are not valid UTF-8 the `String` simply holds those bytes (compare with `codeunits`). Input is read as bytes (`codeunits`).
- EOF on `,` stores 0 in the cell.
- The pointer check covers every cell the pointer visits, including cells passed through inside a folded run (for example `<>` at cell 1 throws), because `MOVE` carries the run's extreme offsets.
- The error for `[` that is never closed reports the position of the outermost unmatched `[`; `]` without `[` reports its own position.

## Tests

```julia
using Test

const HELLO = "++++++++++[>+++++++>++++++++++>+++>+<<<<-]>++.>+.+++++++..+++.>++.<<+++++++++++++++.>.+++.------.--------.>+.>."

@testset "brainfuck" begin
    @test run_bf(HELLO) == "Hello World!\n"
    @test run_bf(",[.,]", "echo me") == "echo me"                  # echo until EOF (0)
    @test run_bf(",>,<[->+<]>.", "12") == "c"                      # '1' + '2' = 49 + 50 = 99
    @test run_bf(">,[>,]<[.<]", "abc") == "cba"                     # reverse
    @test codeunits(run_bf("-.")) == [0xff]                        # wraparound below 0
    @test codeunits(run_bf("-" ^ 257 * ".")) == [0xff]
    @test codeunits(run_bf("+" ^ 256 * ".")) == [0x00]
    @test run_bf("+++++[-].") == "\0"                              # clear
    @test run_bf("+++++[-]+++.") == "\x03"
    @test run_bf("comment +. here") == "\x01"
    @test run_bf("") == ""
    @test run_bf("+[-]>+<[>-<-]>.") == "\x01"                    # [ is skipped on a zero cell
    @test run_bf("[.]") == ""                                      # loop skipped when cell is 0
    @test run_bf(">" ^ 29999 * "+.") == "\x01"                     # last cell is reachable
    @test_throws BFError run_bf("<")
    @test_throws BFError run_bf(">" ^ 30000)
    @test_throws BFError run_bf("<>")                              # transient excursion inside a folded run
    @test_throws "unmatched ']' at position 3" run_bf("+.]")
    @test_throws "unmatched '[' at position 2" run_bf("+[+")
    # compile folds runs and detects clears
    @test length(compile("++++---")) == 1
    @test compile("[-]")[1].op == CLEAR && length(compile("[-]")) == 1
    @test isempty(compile("+-"))
    @test length(compile("[--]")) == 3
end
```

## Speed

```julia
using BenchmarkTools
# three nested loops over cells starting at 255: 255^3 = 16.6 million iterations
# of the innermost loop `>+<-`, about 83 million executed instructions
const HEAVY = "-[>-[>-[>+<-]<-]<-]>>>."
run_bf(HEAVY)
@btime run_bf(HEAVY);
```

Measured (Julia 1.13.2, one thread): `run_bf(HEAVY)` takes 153 ms, i.e. about 83 million executed instructions at roughly 1.8 ns each, with 14 allocations (the tape and the compiled program). Its output is the single byte `0xff` (255^3 mod 256 increments of the third cell). All 23 tests passed. Two notes from running the tests: `[.<]`-style loops that walk left need a zero cell to their left (a left-walking program that starts at cell 1 correctly raises the off-tape error), and a program such as `><` is kept as a `MOVE` with net shift 0 so its bounds check still happens.
