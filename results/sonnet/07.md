Adaptive Simpson with Richardson extrapolation. Each recursion level receives the already-computed `f` values at the interval's endpoints and midpoint plus the whole-interval Simpson estimate, so every level evaluates `f` only at the two new quarter points (no recomputation).

```julia
function integrate(f, a, b; rtol=1e-8, maxdepth=50)
    a, b = promote(a, b)
    T = float(typeof(a))
    a, b = T(a), T(b)
    a == b && return zero(T), zero(T)
    sgn = one(T)
    if a > b
        a, b = b, a
        sgn = -one(T)
    end
    m = (a + b) / 2
    fa, fm, fb = f(a), f(m), f(b)
    whole = (b - a) / 6 * (fa + 4fm + fb)
    # Tolerance scale from an estimate of the integral of |f|: keeps the target
    # meaningful when the integral itself is zero or heavily cancelled.
    scale = (b - a) / 6 * (abs(fa) + 4abs(fm) + abs(fb))
    # A target below the rounding level of T can never be met.
    rt = max(T(rtol), 8 * eps(T))
    hit = Ref(false)
    val, err = _simpson(f, a, b, fa, fm, fb, whole, rt * scale, 1, maxdepth, hit, eps(T))
    if hit[]
        @warn "integrate: maxdepth = $maxdepth reached; the result may not meet rtol = $rtol" a b
    end
    return sgn * val, err
end

function _simpson(f, a, b, fa, fm, fb, whole, tol, depth, maxdepth, hit, ϵ)
    m = (a + b) / 2
    flm = f((a + m) / 2)
    frm = f((m + b) / 2)
    h = b - a
    left  = h / 12 * (fa + 4flm + fm)
    right = h / 12 * (fm + 4frm + fb)
    δ = left + right - whole
    # Accept when the Richardson error estimate |δ|/15 meets the tolerance,
    # or when δ is at the rounding level of the sums.
    if abs(δ) <= 15 * tol || abs(δ) <= 16ϵ * (abs(left) + abs(right))
        return left + right + δ / 15, abs(δ) / 15
    elseif depth >= maxdepth
        hit[] = true
        return left + right + δ / 15, abs(δ) / 15
    end
    vl, el = _simpson(f, a, m, fa, flm, fm, left,  tol / 2, depth + 1, maxdepth, hit, ϵ)
    vr, er = _simpson(f, m, b, fm, frm, fb, right, tol / 2, depth + 1, maxdepth, hit, ϵ)
    return vl + vr, el + er
end
```

Details:

- `(left + right - whole)/15` is the Richardson correction; it is added to the value, and `|δ|/15` summed over the leaves is returned as `errest`.
- The tolerance halves at each level so the leaves' errors add up to at most the global tolerance.
- `a > b` swaps the limits and flips the sign. `a == b` returns `(0, 0)`.
- `Float32`, `Float64`, `BigFloat` all work because every constant is derived from `T`; the `rtol` floor (`8 eps(T)`) matters for `Float32`, where the default `1e-8` is unreachable.
- Reaching `maxdepth` emits an `@warn` once per call (not once per leaf) and returns the best estimate.
- Integer limits are accepted (`float` is applied); `f` should return the same floating type as its argument for best performance.

## Examples

```julia
using Test

for (name, f, a, b, ex) in (("sin [0,π]", sin, 0.0, π, 2.0),
                           ("sqrt [0,1]", sqrt, 0.0, 1.0, 2/3),
                           ("exp(-x²) [-5,5]", x -> exp(-x^2), -5.0, 5.0, sqrt(π)))
    v, e = integrate(f, a, b)
    println(rpad(name, 18), " value=", v, "  errest=", e, "  |error vs exact|=", abs(v - ex))
end
```

Exact references: ∫₀^π sin = 2, ∫₀¹ √x = 2/3, ∫₋₅⁵ exp(-x²) = √π·erf(5) = √π − 2.7e-12 approximately (`erf` is not in Base, so I compare with √π; the difference is far below `rtol = 1e-8`).

## Tests

```julia
@testset "integrate" begin
    for T in (Float32, Float64, BigFloat)
        v, e = integrate(sin, T(0), T(π))
        @test v isa T
        @test isapprox(v, 2; rtol = T == Float32 ? 1e-5 : 1e-7)
    end
    v, _ = integrate(sin, π, 0.0)
    @test v ≈ -2 rtol = 1e-7
    @test integrate(sin, 1.0, 1.0) == (0.0, 0.0)
    @test integrate(x -> x^2, 0, 3)[1] ≈ 9 rtol = 1e-10        # integer limits
    @test integrate(sin, -1.0, 1.0)[1] ≈ 0 atol = 1e-10         # zero integral terminates
    @test integrate(x -> exp(-x^2), -5.0, 5.0)[1] ≈ sqrt(π) rtol = 1e-8
    @test_logs (:warn, r"maxdepth") integrate(x -> sqrt(abs(x)), -1.0, 1.0; rtol = 1e-14, maxdepth = 5)
    # function values are not recomputed: count calls
    n = Ref(0); integrate(x -> (n[] += 1; x), 0.0, 1.0)
    @test n[] == 5          # 3 initial evaluations + 2 quarter points, accepted immediately
end
```
