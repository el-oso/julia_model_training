```julia
"""
    merge_intervals(v) -> Vector{Tuple{T,T}}

Merge closed intervals `(start, stop)` (any `Real` types, any order) into the
minimal sorted list of disjoint intervals covering the same points. Intervals
that touch (`stop == next start`) merge. `v` is not modified. O(n log n).
"""
function merge_intervals(v::AbstractVector{<:Tuple{Real,Real}})
    ET = eltype(v)
    T = isconcretetype(ET) ? promote_type(fieldtype(ET, 1), fieldtype(ET, 2)) :
        mapreduce(t -> promote_type(typeof(t[1]), typeof(t[2])), promote_type, v; init = Union{})   # mixed-type literals
    out = Tuple{T,T}[]
    isempty(v) && return out
    for (a, b) in v
        a <= b || throw(ArgumentError("interval ($a, $b) has start > stop"))
    end
    s = sort(v; by = first)            # a sorted copy; the input is untouched
    lo, hi = s[1]
    lo = convert(T, lo); hi = convert(T, hi)
    for k in firstindex(s)+1:lastindex(s)
        a, b = s[k]
        if a <= hi                     # overlaps or touches the current run
            b > hi && (hi = convert(T, b))
        else
            push!(out, (lo, hi))
            lo = convert(T, a); hi = convert(T, b)
        end
    end
    push!(out, (lo, hi))
    return out
end
```

Notes: the sort is on a copy, so `v` is never mutated. Unsorted ties on `start` are fine because the sweep takes the max of the `stop` values. A `NaN` endpoint fails the `a <= b` check and throws. The element type of the result is the promotion of all endpoint types (computed per element when the input vector has an abstract element type, as for a literal mixing `Int` and `Float64` tuples), so `[(1, 2.5)]` gives `Vector{Tuple{Float64,Float64}}`.

## Tests

```julia
using Test, Random

# Sample-point check. Endpoints are integers, so any gap between merged
# intervals has length >= 1 and contains a half-integer, and testing every
# half-integer detects every difference between the two covers.
covered(ivs, x) = any(iv -> iv[1] <= x <= iv[2], ivs)

@testset "merge_intervals" begin
    @test merge_intervals(Tuple{Int,Int}[]) == Tuple{Int,Int}[]
    @test merge_intervals([(1, 2)]) == [(1, 2)]
    @test merge_intervals([(3, 4), (1, 2)]) == [(1, 2), (3, 4)]          # unsorted, disjoint
    @test merge_intervals([(1, 2), (2, 3)]) == [(1, 3)]                  # touching merges
    @test merge_intervals([(1, 5), (2, 3)]) == [(1, 5)]                  # nested
    @test merge_intervals([(1, 5), (1, 3), (1, 9)]) == [(1, 9)]          # equal starts
    @test merge_intervals([(1, 1), (1, 1)]) == [(1, 1)]                  # degenerate
    @test merge_intervals([(5, 6), (1, 2), (2, 4), (4, 5)]) == [(1, 6)]  # chain
    @test merge_intervals([(0.5, 1.5), (1.5, 2)]) == [(0.5, 2.0)]
    @test merge_intervals([(1, 2.5), (2, 4)]) isa Vector{Tuple{Float64,Float64}}
    @test merge_intervals([(1//2, 1), (1, 3//2)]) == [(1//2, 3//2)]
    @test merge_intervals([(-3, -1), (-2, 4)]) == [(-3, 4)]
    @test merge_intervals(Tuple{UInt8,UInt8}[(0x01, 0x03), (0x03, 0x05)]) == [(0x01, 0x05)]
    @test_throws ArgumentError merge_intervals([(2, 1)])

    # input not mutated
    v = [(5, 6), (1, 2), (2, 4)]
    w = copy(v)
    merge_intervals(v)
    @test v == w

    # randomized property test against a brute-force cover check
    rng = Xoshiro(42)
    for trial in 1:2000
        n = rand(rng, 0:12)
        v = Tuple{Int,Int}[(a = rand(rng, 0:20); (a, a + rand(rng, 0:6))) for _ in 1:n]
        orig = copy(v)
        m = merge_intervals(v)
        @test v == orig
        @test all(i -> m[i][1] <= m[i][2], eachindex(m))
        @test all(i -> m[i][2] < m[i+1][1], 1:length(m)-1)     # sorted, disjoint, not touching (minimal)
            for x in -1.0:0.5:28.0
            @test covered(v, x) == covered(m, x)
        end
        @test merge_intervals(m) == m                          # idempotent
        # every merged interval's endpoints come from the input
        @test all(iv -> any(t -> t[1] == iv[1], v) && any(t -> t[2] == iv[2], v), m)
    end
end
```

All tests above passed when run (128,015 checks, 2,000 random trials).
