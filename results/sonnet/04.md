Approach: Dijkstra with a hand-written binary min-heap and lazy deletion (a node may be pushed several times; stale entries are skipped when popped). The heap is stored as two parallel arrays (keys of weight type `W`, node ids as `Int`) so no tuples are allocated. Sifting uses the "hole" technique (move parents/children into the hole, write the new element once).

```julia
# ---- binary min-heap on parallel arrays ----
function heap_push!(hk::Vector{W}, hv::Vector{Int}, k::W, v::Int) where {W}
    push!(hk, k); push!(hv, v)          # grow; slot contents are overwritten below
    i = length(hk)
    while i > 1
        p = i >> 1
        hk[p] <= k && break
        hk[i] = hk[p]; hv[i] = hv[p]
        i = p
    end
    hk[i] = k; hv[i] = v
    return nothing
end

function heap_pop!(hk::Vector{W}, hv::Vector{Int}) where {W}
    k0, v0 = hk[1], hv[1]
    k = pop!(hk); v = pop!(hv)           # last element, to be re-inserted from the root
    n = length(hk)
    if n > 0
        i = 1
        while true
            c = 2i
            c > n && break
            c < n && hk[c+1] < hk[c] && (c += 1)
            hk[c] >= k && break
            hk[i] = hk[c]; hv[i] = hv[c]
            i = c
        end
        hk[i] = k; hv[i] = v
    end
    return k0, v0
end

"""
    dijkstra(adj, src) -> (dist, prev)

`adj[u]` is a vector of `(v, w)` tuples (edge u -> v, weight w >= 0). The tuple
type must be concrete, e.g. `Vector{Vector{Tuple{Int,Float64}}}`; the weight
type `W` is read from it. Unreachable nodes get `dist = typemax(W)` and
`prev = 0`; `prev[src] == 0`.
"""
function dijkstra(adj::AbstractVector, src::Integer)
    Base.require_one_based_indexing(adj)
    n = length(adj)
    1 <= src <= n || throw(BoundsError(adj, src))
    W = fieldtype(eltype(eltype(adj)), 2)
    dist = fill(typemax(W), n)
    prev = zeros(Int, n)
    dist[src] = zero(W)
    hk = W[]; hv = Int[]
    heap_push!(hk, hv, zero(W), Int(src))
    while !isempty(hk)
        d, u = heap_pop!(hk, hv)
        d > dist[u] && continue           # stale entry
        for (v, w) in adj[u]
            w < 0 && throw(ArgumentError("negative weight $w on edge $u -> $v"))
            nd = d + w
            if nd < dist[v]
                dist[v] = nd
                prev[v] = u
                heap_push!(hk, hv, nd, Int(v))
            end
        end
    end
    return dist, prev
end

"""
    path(prev, target) -> Vector{Int}

Nodes from the source to `target`. If `target` is unreachable the result is
`[target]`; check `dist[target] == typemax(eltype(dist))` first if that matters
(`prev` alone cannot distinguish the source from an unreachable node).
"""
function path(prev::AbstractVector{<:Integer}, target::Integer)
    p = Int[]
    u = Int(target)
    while u != 0
        push!(p, u)
        u = prev[u]
    end
    return reverse!(p)
end
```

Notes:

- Integer weights work (`dist` uses `typemax(Int)` as infinity); sums are not overflow-checked.
- Each heap entry costs `sizeof(W) + 8` bytes and there can be up to one entry per edge relaxation that improved a distance, so the heap is bounded by the number of edges.
- The `(v, w)` loop destructures concretely typed tuples, so the inner loop is allocation-free.

## Tests

```julia
using Test, Random

@testset "dijkstra" begin
    adj = [[(2, 7), (3, 9), (6, 14)], [(1, 7), (3, 10), (4, 15)],
           [(1, 9), (2, 10), (4, 11), (6, 2)], [(2, 15), (3, 11), (5, 6)],
           [(4, 6), (6, 9)], [(1, 14), (3, 2), (5, 9)], Tuple{Int,Int}[]]
    dist, prev = dijkstra(adj, 1)
    @test dist[1:6] == [0, 7, 9, 20, 20, 11]
    @test dist[7] == typemax(Int) && prev[7] == 0
    @test path(prev, 5) == [1, 3, 6, 5]
    @test path(prev, 1) == [1]

    # Float weights vs. Bellman-Ford on random graphs
    rng = Xoshiro(1)
    for _ in 1:50
        n = rand(rng, 2:30)
        g = [Tuple{Int,Float64}[] for _ in 1:n]
        for _ in 1:3n
            push!(g[rand(rng, 1:n)], (rand(rng, 1:n), rand(rng)))
        end
        d, _ = dijkstra(g, 1)
        bf = fill(Inf, n); bf[1] = 0.0
        for _ in 1:n, u in 1:n, (v, w) in g[u]
            bf[u] + w < bf[v] && (bf[v] = bf[u] + w)
        end
        @test d ≈ bf
    end

    @test_throws ArgumentError dijkstra([[(1, -1.0)]], 1)
end
```

## Benchmark setup (10^6 nodes, 5×10^6 edges)

Measured on Julia 1.13.2, single thread (BenchmarkTools): about 567 ms for the 10^6-node, 5×10^6-edge graph (64 allocations, 49 MiB for the heap and result arrays).
