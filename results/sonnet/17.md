Encoding: `A=0, C=1, G=2, T=3`, 2 bits each. A rolling code is `code = ((code << 2) | base) & mask` with `mask = 4^k - 1`, so each new base costs a shift, an or and an and. `k <= 31` keeps `2k <= 62` bits in a `UInt64` (the shift never overflows into the sign/mask logic). Bases are looked up in a 256-entry table over the string's bytes.

Non-`ACGT` characters, including lowercase and `N`, throw an `ArgumentError` with the byte position. (Documented choice; lowercase is not folded to uppercase.)

```julia
const BASE_CODE = let t = fill(0xff, 256)
    for (i, c) in enumerate("ACGT")
        t[UInt8(c) + 1] = UInt8(i - 1)
    end
    t
end

"""
    count_kmers(seq, k) -> Dict{UInt64,Int}

Count every length-`k` substring (1 <= k <= 31) of the DNA string `seq`,
keyed by its 2-bit code (`A=0 C=1 G=2 T=3`, first base in the high bits).
Throws `ArgumentError` on any character other than uppercase `A C G T`.
"""
function count_kmers(seq::AbstractString, k::Integer)
    1 <= k <= 31 || throw(ArgumentError("k must be in 1:31, got $k"))
    counts = Dict{UInt64,Int}()
    cu = codeunits(seq)
    mask = (UInt64(1) << (2k)) - 1
    code = UInt64(0)
    filled = 0                           # bases accumulated so far, capped at k
    for i in eachindex(cu)
        b = BASE_CODE[cu[i] + 1]
        b == 0xff && throw(ArgumentError("invalid base '$(Char(cu[i]))' at position $i (only uppercase A, C, G, T are allowed)"))
        code = ((code << 2) | b) & mask
        if filled < k - 1
            filled += 1
        else
            counts[code] = get(counts, code, 0) + 1
        end
    end
    return counts
end

"""
    decode(code, k) -> String

Inverse of the encoding used by `count_kmers`.
"""
function decode(code::Integer, k::Integer)
    letters = codeunits("ACGT")
    buf = Vector{UInt8}(undef, k)
    c = UInt64(code)
    for i in k:-1:1
        buf[i] = letters[(c & 0x3) + 1]
        c >>= 2
    end
    return String(buf)
end
```

## Tests and timing

```julia
using Test, Random, BenchmarkTools

@testset "count_kmers" begin
    d = count_kmers("ACGTACGT", 2)
    @test Dict(decode(c, 2) => n for (c, n) in d) == Dict("AC" => 2, "CG" => 2, "GT" => 2, "TA" => 1)
    @test sum(values(d)) == 7
    @test count_kmers("ACG", 4) == Dict{UInt64,Int}()
    @test count_kmers("", 3) == Dict{UInt64,Int}()
    @test decode(count_kmers("TTT", 3) |> only |> first, 3) == "TTT"
    @test decode(0x1b, 4) == "ACGT"              # 00 01 10 11

    # against slicing, with k up to 31
    rng = Xoshiro(1)
    for k in (1, 5, 17, 31)
        s = randstring(rng, "ACGT", 500)
        ref = Dict{String,Int}()
        for i in 1:length(s)-k+1
            ref[s[i:i+k-1]] = get(ref, s[i:i+k-1], 0) + 1
        end
        got = Dict(decode(c, k) => n for (c, n) in count_kmers(s, k))
        @test got == ref
    end
    @test_throws ArgumentError count_kmers("ACGT", 0)
    @test_throws ArgumentError count_kmers("ACGT", 32)
    @test_throws "position 3" count_kmers("ACnT", 2)
    @test_throws ArgumentError count_kmers("acgt", 2)       # lowercase is rejected
end

seq = randstring(Xoshiro(2), "ACGT", 10^7)
for k in (5, 12, 21, 31)
    count_kmers(seq, k)
    println("k = $k: ", round(minimum(@elapsed(count_kmers(seq, k)) for _ in 1:3), digits = 3), " s")
end
```

Measured (Julia 1.13.2, one thread, random 10^7-base sequence, minimum of 3 runs after warm-up): k = 5: 0.11 s; k = 12: 0.98 s; k = 21: 0.93 s; k = 31: 0.87 s. For small `k` the dictionary has at most `4^k` keys and stays cache-resident; for large `k` almost every k-mer is distinct, so the time is dominated by about 10^7 dictionary insertions (the rolling update itself is cheap). I did not try a faster table; for `k <= 12` a dense `Vector{Int}` of length `4^k` would avoid hashing, but the required return type is a `Dict`.
