```julia
# Run with several threads, e.g.  julia -t 8 wordfreq.jl DIR
function count_file(path::AbstractString)
    counts = Dict{String,Int}()
    text = read(path, String)
    start = 0                       # byte index where the current word starts, 0 = not in a word
    for (i, c) in pairs(text)
        if isletter(c)
            start == 0 && (start = i)
        elseif start != 0
            w = lowercase(SubString(text, start, prevind(text, i)))
            counts[w] = get(counts, w, 0) + 1
            start = 0
        end
    end
    if start != 0
        w = lowercase(SubString(text, start))
        counts[w] = get(counts, w, 0) + 1
    end
    return counts
end

function count_words(dir::AbstractString)
    files = filter(f -> endswith(f, ".txt") && isfile(f), readdir(dir; join=true))
    tasks = [Threads.@spawn count_file(f) for f in files]   # one task per file, each owns its Dict
    total = Dict{String,Int}()
    for t in tasks                                           # merge on this task only
        for (w, n) in fetch(t)
            total[w] = get(total, w, 0) + n
        end
    end
    return total
end

function top_words(total, n=20)
    # frequency descending, ties broken alphabetically for deterministic output
    sorted = sort!(collect(total); by = p -> (-p.second, p.first))
    return first(sorted, n)
end

function main(dir = isempty(ARGS) ? "." : ARGS[1])
    for (w, n) in top_words(count_words(dir))
        println(rpad(w, 20), n)
    end
end

if abspath(PROGRAM_FILE) == @__FILE__
    main()
end
```

## Why the merge is race-free

- Each spawned task creates its own `Dict` inside `count_file` and is the only code that ever touches it while it runs. There is no shared mutable state during counting.
- The merged dictionary `total` is only written by the task that called `count_words` (the loop over `fetch(t)`). `fetch` waits for the task to finish and establishes a happens-before edge, so the worker's writes to its `Dict` are completely visible, and after a task finishes nobody else mutates its `Dict`.
- So at any moment, every mutable object has exactly one writer. No locks or atomics are required. (A shared `Dict` with `+=` from several tasks would be a data race: `Dict` is not thread-safe, and the read-modify-write on a count is not atomic even with a lock-free structure.)

## How it scales

Measured: 16 generated files (61 MB total, about 600k words per file, vocabulary 20,000 random words), 6 pinned cores, minimum of 3 runs after warm-up: 1 thread 2.61 s, 2 threads 1.52 s, 4 threads 0.83 s, 6 threads 0.74 s (3.5x on 6 threads). The rest below is what the structure implies and was not separately measured:

- Parallelism is per file: with `F` files and `T` threads the useful speedup is at most `min(F, T)`, and one very large file dominates the wall time (a single task counts it serially). To balance a few large files, split them into byte chunks at whitespace boundaries and spawn a task per chunk (the merge step is unchanged).
- The merge is serial and costs about the total number of distinct words summed over files, not the number of word occurrences; it is small relative to counting when the vocabulary is much smaller than the text, and it becomes the serial fraction (Amdahl) when files are many and vocabularies large and mostly disjoint. A tree-merge or merging in the spawned tasks via `reduce` of pairs would shorten it.
- Counting allocates one `String` per word occurrence (`lowercase` of a substring), so the garbage collector is shared across threads and can limit scaling; reducing allocations (for example counting with `SubString` keys of a lowercased copy, or interning) is the first thing to try if profiling shows GC time.
- Start Julia with `-t N` for `N` below the core count to leave cores free.
