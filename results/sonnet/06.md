```julia
struct Windows{V<:AbstractVector,E}
    v::V
    k::Int
    function Windows(v::V, k::Integer) where {V<:AbstractVector}
        k >= 1 || throw(ArgumentError("window length must be positive, got $k"))
        E = Base.promote_op(view, V, UnitRange{Int})
        new{V,E}(v, Int(k))
    end
end

windows(v::AbstractVector, k::Integer) = Windows(v, k)

Base.length(w::Windows) = max(0, length(w.v) - w.k + 1)
Base.eltype(::Type{<:Windows{V,E}}) where {V,E} = E
Base.IteratorSize(::Type{<:Windows}) = Base.HasLength()

# The state is the first index of the next window.
@inline function Base.iterate(w::Windows, i::Int = firstindex(w.v))
    stop = i + w.k - 1
    stop > lastindex(w.v) && return nothing
    return view(w.v, i:stop), i + 1
end
```

Notes:

- The state is an `Int` start index in the parent's own index space (`firstindex(v)`), so `OffsetArray`s work. Each window is an ordinary `SubArray` view (1-based axes), as `view(v, i:j)` gives for any vector.
- `k > length(v)` gives `length == 0` and `iterate` returns `nothing` immediately; `k <= 0` throws `ArgumentError` at construction.
- The element type is computed by inference (`Base.promote_op`) so `eltype` is concrete and `collect` returns a correctly typed `Vector` even when there are zero windows.
- Each window is a small `SubArray` struct. When the loop body does not let it escape, the compiler keeps it on the stack; the allocation check below verifies that for this case.

## Tests and benchmark

```julia
using Test, BenchmarkTools

@testset "windows" begin
    v = collect(1:6)
    @test length(windows(v, 3)) == 4
    @test collect(windows(v, 3)) == [[1,2,3],[2,3,4],[3,4,5],[4,5,6]]
    @test all(w -> w isa SubArray, windows(v, 3))
    @test eltype(windows(v, 3)) == typeof(view(v, 1:3))
    @test isempty(windows(v, 7)) && length(windows(v, 7)) == 0
    @test collect(windows(v, 7)) == Vector{Int}[] || isempty(collect(windows(v, 7)))
    @test length(windows(v, 6)) == 1
    @test length(windows(Int[], 1)) == 0
    @test_throws ArgumentError windows(v, 0)
    @test_throws ArgumentError windows(v, -2)
    @test length(collect(windows(view(v, 2:5), 2))) == 3
    @test collect(windows(1:5, 2)) == [1:2, 2:3, 3:4, 4:5]   # views of ranges
    # windows are views: writing through one changes the parent
    x = collect(1:5); first(windows(x, 2))[1] = 100; @test x[1] == 100
end

function count_sum(w)
    s = 0
    for win in w
        s += win[1]
    end
    s
end
@testset "no per-window allocation" begin
    v = rand(1000)
    f(v) = (s = 0.0; for w in windows(v, 16); s += w[1]; end; s)
    f(v)
    @test (@allocated f(v)) == 0
end

v = rand(10^7)
@show sum(sum, windows(v, 16))
@btime sum(sum, windows($v, 16));
```

Measured (Julia 1.13.2, single thread, BenchmarkTools): `sum(sum, windows(v, 16))` on 10^7 `Float64` takes 71.6 ms with 0 allocations; the result for the random input was about 8.0e7 (it varies per run). The `for` loop test confirms zero allocation per window. `OffsetArray` support follows from using `firstindex`/`lastindex`, but OffsetArrays was not installed here, so that case was not run.
