```julia
using LinearAlgebra: SingularException

"""
    lu_pp!(A) -> (A, p)

In-place LU with partial pivoting of a square matrix. On return `A` holds `U`
on and above the diagonal and the multipliers of the unit lower-triangular `L`
below it; `p` is the row permutation with `A_original[p, :] == L * U`.
Throws `SingularException(k)` if no nonzero pivot exists in column `k`.
The element type needs `abs`, `/`, `*`, `-` (so `Float64`, `Rational{BigInt}`, ...);
integer matrices are not supported because the multipliers are quotients.
"""
function lu_pp!(A::AbstractMatrix)
    Base.require_one_based_indexing(A)
    n = size(A, 1)
    size(A, 2) == n || throw(DimensionMismatch("matrix must be square, got $(size(A))"))
    p = collect(1:n)
    for k in 1:n
        # pivot: largest |A[i,k]|, i >= k
        piv = k
        mx = abs(A[k, k])
        for i in k+1:n
            v = abs(A[i, k])
            if v > mx
                mx = v; piv = i
            end
        end
        iszero(mx) && throw(SingularException(k))
        if piv != k
            for j in 1:n
                A[k, j], A[piv, j] = A[piv, j], A[k, j]
            end
            p[k], p[piv] = p[piv], p[k]
        end
        akk = A[k, k]
        for i in k+1:n
            A[i, k] /= akk
        end
        # rank-1 update of the trailing block, column by column (inner loop over i is contiguous)
        for j in k+1:n
            akj = A[k, j]
            for i in k+1:n
                A[i, j] -= A[i, k] * akj
            end
        end
    end
    return A, p
end

function lu_pp(A::AbstractMatrix)
    T = typeof(one(eltype(A)) / one(eltype(A)))     # Int -> Float64, Rational stays Rational
    F, p = lu_pp!(Matrix{T}(A))      # always a copy
    n = size(F, 1)
    L = [i > j ? F[i, j] : (i == j ? one(T) : zero(T)) for i in 1:n, j in 1:n]
    U = [i <= j ? F[i, j] : zero(T) for i in 1:n, j in 1:n]
    return L, U, p
end

function lu_solve(L::AbstractMatrix, U::AbstractMatrix, p::AbstractVector{<:Integer}, b::AbstractVector)
    n = size(L, 1)
    (size(L) == (n, n) && size(U) == (n, n) && length(p) == n && length(b) == n) ||
        throw(DimensionMismatch("inconsistent sizes"))
    T = promote_type(eltype(L), eltype(U), eltype(b))
    x = T[b[p[i]] for i in 1:n]
    for j in 1:n-1                   # forward: L has a unit diagonal
        xj = x[j]
        for i in j+1:n
            x[i] -= L[i, j] * xj
        end
    end
    for j in n:-1:1                  # backward
        x[j] /= U[j, j]
        xj = x[j]
        for i in 1:j-1
            x[i] -= U[i, j] * xj
        end
    end
    return x
end
```

Notes:

- Row swaps swap entire rows (including the already computed multipliers), the standard storage scheme, so `A[p, :] == L*U` holds exactly in exact arithmetic.
- Singularity is exact-zero pivot detection. For floating point, a numerically singular matrix usually yields tiny pivots rather than exactly zero; `Rational` types detect singularity exactly. `lu_pp!` is called with the same loops regardless of element type, so it is generic; inner loops are column-major.
- The pivot selection uses `abs`, which for `Rational` picks the largest numerator/denominator quotient; any nonzero pivot would be exact, so for `Rational` this is only for consistency.
- No `@inbounds`: I measured the trailing-update loop with `@inbounds @simd` and it changed the n = 500 time from 9.09 ms to 8.92 ms (about 2%), so the checks are not the bottleneck and the annotation is not worth its risk here.

## Tests and comparison

```julia
using Test, LinearAlgebra, BenchmarkTools, Random

@testset "lu_pp" begin
    rng = Xoshiro(3)
    for n in (1, 2, 5, 30)
        A = randn(rng, n, n)
        L, U, p = lu_pp(A)
        @test A[p, :] ≈ L * U
        @test istril(L) && istriu(U) && all(==(1), diag(L))
        b = randn(rng, n)
        @test lu_solve(L, U, p, b) ≈ A \ b
    end
    # exact arithmetic
    A = Rational{BigInt}.(rand(rng, -5:5, 6, 6))
    while iszero(det(Float64.(A))); A = Rational{BigInt}.(rand(rng, -5:5, 6, 6)); end
    L, U, p = lu_pp(A)
    @test A[p, :] == L * U                       # exact equality
    b = Rational{BigInt}.(rand(rng, -5:5, 6))
    x = lu_solve(L, U, p, b)
    @test eltype(x) == Rational{BigInt} && A * x == b
    # needs pivoting
    P = [0.0 1.0; 1.0 0.0]
    L, U, p = lu_pp(P); @test p == [2, 1] && P[p, :] == L * U
    # integer input goes to Float64
    @test lu_pp([2 1; 1 3])[2] isa Matrix{Float64}
    # singular
    @test_throws SingularException lu_pp(ones(3, 3))
    @test_throws SingularException lu_pp(Rational{BigInt}[1 2; 2 4])
    @test_throws DimensionMismatch lu_pp!(ones(2, 3))
    # in place
    B = copy(A); F, q = lu_pp!(B); @test F === B
end

n = 500
A = randn(Xoshiro(1), n, n)
```

## Measured at n = 500 (Julia 1.13.2, Float64, BenchmarkTools)

| | time |
|---|---|
| `lu_pp(A)` (copies, builds L and U) | 9.6 ms |
| `lu_pp!(B)` (in place, no L/U extraction) | 9.2 ms |
| `lu(A)` from LinearAlgebra, `OPENBLAS_NUM_THREADS=1` | 2.5 ms |
| `lu(A)`, default OpenBLAS threads (6 cores available) | 1.9 ms |

So this straightforward right-looking version is about 3.7x slower than LAPACK single-threaded (4.8x against the multithreaded run). I did not profile the cause. LAPACK's LU is a blocked (recursive in OpenBLAS) algorithm, whereas this version sweeps the whole trailing matrix once per step `k`; that difference is the likely reason, not something I measured. A blocked variant would be the next step; I did not write one. The tests (random Float64 at several sizes, exact `Rational{BigInt}` equality for `A[p,:] == L*U` and `A*x == b`, a matrix that needs pivoting, integer input, singular inputs, in-place identity) all passed.
