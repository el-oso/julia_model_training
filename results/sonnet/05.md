```julia
struct Interval{T<:Real} <: Real
    lo::T
    hi::T
    function Interval{T}(lo, hi) where {T<:Real}
        lo <= hi || throw(ArgumentError("invalid interval: lo = $lo > hi = $hi"))
        new{T}(lo, hi)
    end
end

Interval(lo::Real, hi::Real) = Interval{promote_type(typeof(lo), typeof(hi))}(lo, hi)
Interval(x::Real) = Interval(x, x)
Interval(x::Interval) = x

# ---- promotion / conversion ----
Base.promote_rule(::Type{Interval{T}}, ::Type{Interval{S}}) where {T<:Real,S<:Real} =
    Interval{promote_type(T, S)}
Base.promote_rule(::Type{Interval{T}}, ::Type{S}) where {T<:Real,S<:Real} =
    Interval{promote_type(T, S)}

Base.convert(::Type{Interval{T}}, x::Interval{T}) where {T<:Real} = x
Base.convert(::Type{Interval{T}}, x::Interval) where {T<:Real} = Interval{T}(x.lo, x.hi)
Base.convert(::Type{Interval{T}}, x::Real) where {T<:Real} = Interval{T}(x, x)

# ---- arithmetic (same element type; mixed operands go through promotion) ----
Base.:+(a::Interval{T}, b::Interval{T}) where {T} = Interval(a.lo + b.lo, a.hi + b.hi)
Base.:-(a::Interval{T}, b::Interval{T}) where {T} = Interval(a.lo - b.hi, a.hi - b.lo)
Base.:-(a::Interval) = Interval(-a.hi, -a.lo)

function Base.:*(a::Interval{T}, b::Interval{T}) where {T}
    p1 = a.lo * b.lo; p2 = a.lo * b.hi; p3 = a.hi * b.lo; p4 = a.hi * b.hi
    return Interval(min(p1, p2, p3, p4), max(p1, p2, p3, p4))
end

function Base.:/(a::Interval{T}, b::Interval{T}) where {T}
    b.lo <= 0 <= b.hi &&
        throw(DomainError(b, "division by an interval containing zero"))
    return a * Interval(inv(b.hi), inv(b.lo))   # 1/b is [1/hi, 1/lo] when 0 is not in b
end

Base.in(x::Real, a::Interval) = a.lo <= x <= a.hi
Base.:(==)(a::Interval, b::Interval) = a.lo == b.lo && a.hi == b.hi
Base.hash(a::Interval, h::UInt) = hash(a.hi, hash(a.lo, hash(:Interval, h)))

function Base.show(io::IO, a::Interval)
    print(io, "Interval(")
    show(io, a.lo); print(io, ", "); show(io, a.hi)
    print(io, ")")
end
```

Design notes:

- The inner constructor takes unconstrained arguments and lets `new{T}` coerce them, so `Interval{Float64}(1, 2)` works; the outer `Interval(lo, hi)` only computes `T` by promotion and delegates.
- `Interval(1, 2) + 1.5`: `promote_rule(Interval{Int}, Float64)` gives `Interval{Float64}`, `1.5` converts to the degenerate `[1.5, 1.5]`, and the result is `Interval(2.5, 3.5)`.
- `Interval(1, 2) * Interval(0.5, 1.0)`: both are promoted to `Interval{Float64}`, giving `Interval(0.5, 2.0)`.
- Only same-type methods are defined; Base's `Number` fallbacks (`+(x::Number, y::Number) = +(promote(x, y)...)`) route mixed operands to them.
- `Interval{Float64}` is an `isbits` struct and the methods only use field reads, arithmetic and `min`/`max` of scalars, so operations should not allocate (tested below).
- Limitation: the arithmetic is not outward rounded. A rigorous interval library must round `lo` down and `hi` up; here floating-point results are only the exact-arithmetic interval up to rounding. If an operation yields `NaN` (for example `Inf * 0`), the constructor's `lo <= hi` check throws `ArgumentError`.
- `/` by an interval with a nonzero real-valued denominator `b` of type `Interval{Int}` yields `Float64` bounds (`inv(::Int)` is a float), so integer interval quotients are `Interval{Float64}`.

## Tests

```julia
using Test

@testset "Interval" begin
    @test Interval(1, 2) isa Interval{Int}
    @test Interval(1, 2.5) isa Interval{Float64}
    @test_throws ArgumentError Interval(2, 1)
    @test_throws ArgumentError Interval{Float64}(3.0, 1.0)

    r = Interval(1, 2) + 1.5
    @test r isa Interval{Float64}
    @test r == Interval(2.5, 3.5)

    m = Interval(1, 2) * Interval(0.5, 1.0)
    @test m isa Interval{Float64}
    @test m == Interval(0.5, 2.0)

    @test Interval(1, 2) + Interval(3, 4) == Interval(4, 6)
    @test Interval(1, 2) - Interval(3, 5) == Interval(-4, -1)
    @test -Interval(1, 2) == Interval(-2, -1)
    @test Interval(-2, 3) * Interval(-1, 4) == Interval(-8, 12)
    @test 2 * Interval(1, 2) == Interval(2, 4)
    @test Interval(1.0, 2.0) / Interval(2.0, 4.0) == Interval(0.25, 1.0)
    @test Interval(1.0, 2.0) / 2 == Interval(0.5, 1.0)
    @test Interval(-3.0, -1.0) / Interval(-2.0, -1.0) == Interval(0.5, 3.0)
    @test_throws DomainError Interval(1.0, 2.0) / Interval(-1.0, 1.0)
    @test_throws DomainError Interval(1.0, 2.0) / Interval(0.0, 1.0)
    @test_throws DomainError Interval(1.0, 2.0) / 0

    @test 1.5 in Interval(1, 2)
    @test 1 in Interval(1, 2) && 2 in Interval(1, 2)   # closed
    @test !(2.1 in Interval(1, 2))

    @test sprint(show, Interval(1, 2.5)) == "Interval(1.0, 2.5)"
    @test promote_type(Interval{Int}, Float64) == Interval{Float64}
    @test promote_type(Interval{Int}, Interval{Float32}) == Interval{Float32}
    @test convert(Interval{Float64}, 3) == Interval(3.0, 3.0)
end

@testset "Interval{Float64} does not allocate" begin
    a = Interval(1.0, 2.0); b = Interval(0.5, 3.0); c = Interval(1.0, 4.0)
    function kernel(a, b, c)
        (a + b, a - b, a * b, a / c, -a, 1.5 in a)
        return nothing
    end
    allocs(a, b, c) = (kernel(a, b, c); @allocated kernel(a, b, c))
    allocs(a, b, c)            # warm up
    @test allocs(a, b, c) == 0
end
```

All tests above passed when run, including the zero-allocation test on `Interval{Float64}`. Not outward rounded, as noted above.
