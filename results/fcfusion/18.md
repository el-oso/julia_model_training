```julia
function mymul!(C, A, B)
    m, n = size(C)
    pa, qa = size(A)
    pb, qb = size(B)
    if pa != m || qa != pb || qb != n
        throw(DimensionMismatch("incompatible dimensions"))
    end
    fill!(C, zero(eltype(C)))
    BLOCK = 128
    for i0 in 1:BLOCK:m
        i1 = min(i0 + BLOCK - 1, m)
        for j0 in 1:BLOCK:n
            j1 = min(j0 + BLOCK - 1, n)
            for k0 in 1:BLOCK:qa
                k1 = min(k0 + BLOCK - 1, qa)
                for j = j0:j1
                    for i = i0:i1
                        s = zero(eltype(C))
                        for k = k0:k1
                            s += A[i, k] * B[k, j]
                        end
                        C[i, j] += s
                    end
                end
            end
        end
    end
    return C
end
```
